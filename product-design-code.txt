'use client';

import { cn } from '@/design-system/utils/cn';

export interface ConditionSelectorProps {
  selectedCondition: 'New' | 'Stock' | 'TakeOff';
  onChangeCondition: (cond: 'New' | 'Stock' | 'TakeOff') => void;
  sellersGroup: any;
}

export function ConditionSelector({ selectedCondition, onChangeCondition, sellersGroup }: ConditionSelectorProps) {
  const isNewAvailable = sellersGroup.newOnline.length > 0 || sellersGroup.newLocal.length > 0;
  const isStockAvailable = sellersGroup.stockOnline.length > 0 || sellersGroup.stockLocal.length > 0;
  const isTakeOffAvailable = sellersGroup.takeOffOnline.length > 0 || sellersGroup.takeOffLocal.length > 0;

  const hasAnyCondition = isNewAvailable || isStockAvailable || isTakeOffAvailable;
  if (!hasAnyCondition) return null;

  const baseBtn = 'px-5 py-2 text-xs font-bold border border-border rounded-md transition-colors outline-none';
  const activeBtn = 'border-shop/60 bg-shop/5 text-shop ring-1 ring-shop/30';
  const idleBtn = 'bg-card text-foreground hover:border-shop/40';

  return (
    <div className="mt-2 flex w-full select-none flex-col gap-2.5">
      <span className="text-xs font-bold text-foreground">نوع قطعه را انتخاب کنید:</span>
      <div className="flex flex-wrap gap-2">
        {isNewAvailable ? (
          <button
            type="button"
            onClick={() => onChangeCondition('New')}
            className={cn(baseBtn, selectedCondition === 'New' ? activeBtn : idleBtn)}
          >
            قطعه نو
          </button>
        ) : null}

        {isStockAvailable ? (
          <button
            type="button"
            onClick={() => onChangeCondition('Stock')}
            className={cn(baseBtn, selectedCondition === 'Stock' ? activeBtn : idleBtn)}
          >
            قطعه استوک
          </button>
        ) : null}

        {isTakeOffAvailable ? (
          <button
            type="button"
            onClick={() => onChangeCondition('TakeOff')}
            className={cn(baseBtn, selectedCondition === 'TakeOff' ? activeBtn : idleBtn)}
          >
            قطعه زیرصفری
          </button>
        ) : null}
      </div>
    </div>
  );
}

'use client';

import * as React from 'react';
import { createPortal } from 'react-dom';
import { motion, AnimatePresence } from 'framer-motion';
import { MapPin, X, AlertTriangle, Phone, Navigation, ArrowRight, Store, Loader2 } from 'lucide-react';
import { cn } from '@/design-system/utils/cn';
import { Button } from '@/components/primitives/Button/Button';
import { useGetReportSubjects, useSubmitShopReport } from '@/domains/front/shop/hooks/shop.hooks';
import { Select } from '@/components/primitives/Select/Select';
import { TextArea } from '@/components/primitives/TextArea/TextArea';
import { showToast } from '@/core/utils/toast';
import { Modal, ModalHeader, ModalTitle, ModalBody } from '@/components/composites/Modal';
import { getFullUrl } from '@/core/utils/formatters';

export interface InPersonSellersMapModalProps {
  isOpen: boolean;
  onClose: () => void;
  sellers: any[];
  productTitle: string;
  initialActiveSellerId?: string | null;
}

export function InPersonSellersMapModal({
  isOpen,
  onClose,
  sellers = [],
  productTitle,
  initialActiveSellerId = null,
}: InPersonSellersMapModalProps) {
  const [mounted, setMounted] = React.useState(false);
  const [mapLoaded, setMapLoaded] = React.useState(false);
  const [activeSellerId, setActiveSellerId] = React.useState<string | null>(null);
  const [reportShop, setReportShop] = React.useState<any | null>(null);

  const [selectedSubjectId, setSelectedSubjectId] = React.useState('');
  const [reportDescription, setReportDescription] = React.useState('');

  const mapContainerRef = React.useRef<HTMLDivElement>(null);
  const mapInstanceRef = React.useRef<any>(null);
  const markersRef = React.useRef<Record<string, any>>({});
  const cardRefs = React.useRef<Record<string, HTMLDivElement | null>>({});

  const { data: reportSubjects = [] } = useGetReportSubjects('ShopProductReport');
  const submitReport = useSubmitShopReport();

  React.useEffect(() => {
    setMounted(true);
  }, []);

  React.useEffect(() => {
    if (!isOpen) return;
    if ((window as any).L) {
      setMapLoaded(true);
      return;
    }
    const existingLink = document.getElementById('leaflet-css');
    if (!existingLink) {
      const link = document.createElement('link');
      link.id = 'leaflet-css';
      link.rel = 'stylesheet';
      link.href = 'https://unpkg.com/leaflet@1.9.4/dist/leaflet.css';
      document.head.appendChild(link);
    }
    const existingScript = document.getElementById('leaflet-js');
    if (existingScript) {
      setMapLoaded(true);
      return;
    }
    const script = document.createElement('script');
    script.id = 'leaflet-js';
    script.src = 'https://unpkg.com/leaflet@1.9.4/dist/leaflet.js';
    script.onload = () => setMapLoaded(true);
    document.body.appendChild(script);
  }, [isOpen]);

  React.useEffect(() => {
    if (!isOpen) return;
    document.body.style.overflow = 'hidden';
    const handlePopState = () => onClose();
    window.history.pushState({ modalOpen: 'in-person-map' }, '');
    window.addEventListener('popstate', handlePopState);
    return () => {
      document.body.style.overflow = '';
      window.removeEventListener('popstate', handlePopState);
    };
  }, [isOpen, onClose]);

  React.useEffect(() => {
    if (!mapLoaded || !mapContainerRef.current || !isOpen || sellers.length === 0) return;
    if (mapInstanceRef.current) {
      mapInstanceRef.current.remove();
      mapInstanceRef.current = null;
      markersRef.current = {};
    }
    const L = (window as any).L;
    const targetSellerId = initialActiveSellerId || sellers[0]?.id;
    const targetSeller = sellers.find((s) => s.id === targetSellerId) || sellers[0];
    const initialLat = targetSeller?.shop?.latitude || 35.6892;
    const initialLon = targetSeller?.shop?.longitude || 51.389;
    const map = L.map(mapContainerRef.current, {
      center: [initialLat, initialLon],
      zoom: 12,
      zoomControl: false,
    });
    L.tileLayer('https://{s}.tile.openstreetmap.org/{z}/{x}/{y}.png', {
      attribution: '© OpenStreetMap contributors',
    }).addTo(map);
    L.control.zoom({ position: 'bottomright' }).addTo(map);
    mapInstanceRef.current = map;
    if (initialActiveSellerId) setActiveSellerId(initialActiveSellerId);

    sellers.forEach((seller) => {
      if (!seller.shop?.latitude || !seller.shop?.longitude) return;
      const customIcon = L.divIcon({
        className: 'custom-leaflet-marker',
        html: `<div class="w-8 h-8 bg-primary text-primary-foreground rounded-full flex items-center justify-center border-2 border-background shadow-lg transform -translate-x-1/2 -translate-y-1/2 cursor-pointer transition-transform hover:scale-110"><svg xmlns="http://www.w3.org/2000/svg" width="16" height="16" viewBox="0 0 24 24" fill="none" stroke="currentColor" stroke-width="2.5" stroke-linecap="round" stroke-linejoin="round"><path d="M20 10c0 6-8 12-8 12s-8-6-8-12a8 8 0 0 1 16 0Z"/><circle cx="12" cy="10" r="3"/></svg></div>`,
        iconSize: [32, 32],
        iconAnchor: [16, 16],
      });
      const marker = L.marker([seller.shop.latitude, seller.shop.longitude], { icon: customIcon })
        .addTo(map)
        .on('click', () => {
          setActiveSellerId(seller.id);
          map.flyTo([seller.shop.latitude, seller.shop.longitude], 15);
        });
      markersRef.current[seller.id] = marker;
    });

    return () => {
      if (mapInstanceRef.current) {
        mapInstanceRef.current.remove();
        mapInstanceRef.current = null;
        markersRef.current = {};
      }
    };
  }, [mapLoaded, isOpen, sellers, initialActiveSellerId]);

  React.useEffect(() => {
    if (isOpen && activeSellerId && cardRefs.current[activeSellerId]) {
      cardRefs.current[activeSellerId]?.scrollIntoView({
        behavior: 'smooth',
        block: 'nearest',
        inline: 'center',
      });
    }
  }, [activeSellerId, isOpen]);

  const handleSelectSeller = (seller: any) => {
    setActiveSellerId(seller.id);
    if (mapInstanceRef.current) {
      mapInstanceRef.current.flyTo([seller.shop.latitude, seller.shop.longitude], 15);
    }
  };

  const handleReportSubmit = async () => {
    if (!selectedSubjectId) {
      showToast.error('لطفاً دلیل گزارش را انتخاب کنید');
      return;
    }
    try {
      await submitReport.mutateAsync({
        shopId: reportShop.shop.id,
        shopProductId: reportShop.id,
        reportSubjectId: selectedSubjectId,
        description: reportDescription,
      });
      showToast.success('گزارش خطای قیمت با موفقیت ثبت شد و توسط پشتیبانی بررسی خواهد گردید');
      setReportShop(null);
      setSelectedSubjectId('');
      setReportDescription('');
    } catch (error: any) {
      showToast.error(error.userMessage || 'خطا در ثبت گزارش');
    }
  };

  const formatPrice = (value: number) => new Intl.NumberFormat('fa-IR').format(value);

  if (!mounted || !isOpen) return null;

  return createPortal(
    <div className="fixed inset-0 z-modal flex h-screen w-screen items-center justify-center overflow-hidden">
      <AnimatePresence>
        <motion.div
          initial={{ opacity: 0 }}
          animate={{ opacity: 1 }}
          exit={{ opacity: 0 }}
          onClick={onClose}
          className="fixed inset-0 bg-foreground/40 backdrop-blur-sm"
        />
      </AnimatePresence>

      <motion.div
        initial={{ y: '100%', opacity: 0 }}
        animate={{ y: '0%', opacity: 1 }}
        exit={{ y: '100%', opacity: 0 }}
        transition={{ type: 'spring', stiffness: 350, damping: 35 }}
        className="relative z-10 flex h-screen w-screen flex-col bg-card shadow-lg"
      >
        <div className="flex shrink-0 items-center justify-between border-b border-border bg-muted/30 px-4 py-4">
          <div className="flex items-center gap-2">
            <button
              type="button"
              onClick={onClose}
              aria-label="بازگشت"
              className="-me-1 rounded-full p-1 hover:bg-muted"
            >
              <ArrowRight aria-hidden="true" className="h-5 w-5 text-foreground" />
            </button>
            <span className="flex items-center gap-1.5 text-sm font-bold text-foreground">
              <MapPin aria-hidden="true" className="h-4 w-4 text-primary" />

          </div>
          <button
            type="button"

            aria-label="بستن"
            className="flex items-center justify-center rounded-full p-1.5 transition-colors hover:bg-muted"
          >
            <X aria-hidden="true" className="h-5 w-5" />
          </button>
        </div>

        <div className="relative flex min-h-0 w-full flex-1 flex-col md:flex-row">
          <div className="hidden min-h-0 w-[380px] shrink-0 flex-col gap-3 overflow-y-auto border-e border-border bg-card p-4 md:flex">
            {sellers.map((seller) => {
              const isSelected = activeSellerId === seller.id;
              return (
                <div
                  key={seller.id}
                  onClick={() => handleSelectSeller(seller)}
                  className={cn(
                    'flex h-fit cursor-pointer flex-col overflow-hidden rounded-md border transition-colors',
                    isSelected
                      ? 'border-primary bg-accent/50 p-4 gap-3'
                      : 'border-border bg-card p-3.5 gap-0 hover:border-primary/40'
                  )}
                >
                  <div className="flex w-full items-start gap-3">
                    <div className="flex h-10 w-10 shrink-0 items-center justify-center overflow-hidden rounded-md border border-border bg-muted">
                      {seller.shop.logo ? (
                        <img src={getFullUrl(seller.shop.logo)} className="h-full w-full object-contain" alt="" />
                      ) : (

            <div className="mt-4 flex w-full gap-2.5">

                className="h-10 flex-1 text-xs font-bold"

            </div>

        </Modal>
      </motion.div>
    </div>,
    document.body
  );
}

function LottiePlayer({ path, className = 'w-48 h-48', loop = true }: { path: string; className?: string; loop?: boolean }) {
  const containerRef = React.useRef<HTMLDivElement>(null);

  React.useEffect(() => {
    let anim: any = null;
    let isMounted = true;

    const init = async () => {
      try {
        const altPath = path.includes('/anumations/')
          ? path.replace('/anumations/', '/animations/')
          : path.replace('/animations/', '/anumations/');
        let animationData: any = null;
        for (const p of [path, altPath]) {
          try {
            const res = await fetch(p);
            if (res.ok) {
              animationData = await res.json();
              break;
            }
          } catch {}
        }
        if (!animationData || !isMounted || !containerRef.current) return;
        const lottieModule = await import('lottie-web');
        const lottie = lottieModule.default || lottieModule;
        if (containerRef.current && isMounted) {
          anim = lottie.loadAnimation({
            container: containerRef.current,
            renderer: 'svg',
            loop,
            autoplay: true,
            animationData,
          });
        }
      } catch {}
    };

    init();
    return () => {
      isMounted = false;
      anim?.destroy();
    };
  }, [path, loop]);

  return <div ref={containerRef} className={className} />;
}

export function ProductCommentsSection({

  const [rate, setRate] = React.useState(0);
  const [hoverRate, setHoverRate] = React.useState(0);
  const [commentText, setCommentText] = React.useState('');
  const [isIncognito, setIsIncognito] = React.useState(false);

  const { data: average } = useGetProductCommentsAverage(productId);
  const { data: commentsResponse, isLoading } = useGetProductComments(productId, orderBy, page);
  const createComment = useCreateComment();
  const likeComment = useLikeComment();
  const unlikeComment = useUnlikeComment();

  // استیت محلی برای آپدیت ۱۰۰٪ درجا و آنی لایک و دیس‌لایک
  const [localComments, setLocalComments] = React.useState<any[]>([]);

    if (commentsResponse?.items) {
      setLocalComments(commentsResponse.items);
    }
  }, [commentsResponse]);

    if (e) e.preventDefault();
    if (rate === 0) {
      showToast.error('لطفاً امتیاز خود به این کالا را انتخاب فرمایید');
      return;

      return;
    }
    try {
      await createComment.mutateAsync({ productId, comment: commentText.trim(), rate, isIncognito });
      setRate(0);
      setCommentText('');
      setIsIncognito(false);
      setIsWriteModalOpen(false);
      setIsSuccessModalOpen(true);

      showToast.error(error?.userMessage || 'خطا در ثبت دیدگاه شما');
    }
  };

  // لایک درجا و بدون تاخیر
  const handleLike = (comment: any) => {
    const prev = [...localComments];
    const isLiked = !!comment.userLiked;
    const isDisliked = !!comment.userDisLiked;

    setLocalComments((list) =>
      list.map((c) => {
        if (c.id !== comment.id) return c;
        const curLikes = typeof c.likes === 'object' ? c.likes?.likes : c.likes || 0;
        const curDislikes = typeof c.likes === 'object' ? c.likes?.dislikes : c.dislikes || 0;

        if (isLiked) {
          return { ...c, userLiked: false, likes: Math.max(0, curLikes - 1) };
        } else {
          return {
            ...c,
            userLiked: true,
            userDisLiked: false,
            likes: curLikes + 1,
            dislikes: isDisliked ? Math.max(0, curDislikes - 1) : curDislikes,
          };
        }
      })
    );

    if (isLiked) {
      unlikeComment.mutate(comment.id, {
        onError: () => setLocalComments(prev),
      });
    } else {
      likeComment.mutate(
        { productCommentId: comment.id, likeStatus: 'Like' },
        { onError: () => setLocalComments(prev) }
      );
    }
  };

  // دیس‌لایک درجا و بدون تاخیر

    const prev = [...localComments];

      list.map((c) => {

            ...c,
            userDisLiked: true,
            userLiked: false,

function useCountdown(expiration?: string | null) {
  const [remain, setRemain] = React.useState<{ h: number; m: number; s: number } | null>(null);

  React.useEffect(() => {
    if (!expiration) {
      setRemain(null);
      return;
    }
    const calc = () => {
      const diff = +new Date(expiration) - Date.now();
      if (diff <= 0) {
        setRemain({ h: 0, m: 0, s: 0 });
        return;
      }
      setRemain({
        h: Math.floor(diff / (1000 * 60 * 60)),
        m: Math.floor((diff / 1000 / 60) % 60),
        s: Math.floor((diff / 1000) % 60),
      });
    };
    calc();
    const iv = setInterval(calc, 1000);
    return () => clearInterval(iv);
  }, [expiration]);

  return remain;
}

function PriceChart({ data }: { data: any }) {
  const prices: number[] = data.prices || [];
  if (prices.length === 0) return null;

  const min = Math.min(...prices);
  const max = Math.max(...prices);
  const W = 320;
  const H = 160;
  const pad = 20;
  const innerW = W - pad * 2;
  const innerH = H - pad * 2;

  const points = prices
    .map((p, i) => {
      const x = pad + (i / Math.max(prices.length - 1, 1)) * innerW;
      const y = pad + innerH - ((p - min) / Math.max(max - min, 1)) * innerH;
      return `${x},${y}`;
    })
    .join(' ');

  const fmt = (n: number) => new Intl.NumberFormat('fa-IR').format(n);

  return (
    <div className="w-full px-2">
      <svg viewBox={`0 0 ${W} ${H}`} className="h-40 w-full">
        <line x1={pad} y1={pad} x2={pad} y2={H - pad} stroke="currentColor" className="text-border" strokeWidth="1" />
        <line x1={pad} y1={H - pad} x2={W - pad} y2={H - pad} stroke="currentColor" className="text-border" strokeWidth="1" />
        <polyline
          fill="none"
          stroke="currentColor"
          className="text-shop"
          strokeWidth="2"
          strokeLinejoin="round"
          strokeLinecap="round"
          points={points}
        />
        {prices.map((p, i) => {
          const x = pad + (i / Math.max(prices.length - 1, 1)) * innerW;
          const y = pad + innerH - ((p - min) / Math.max(max - min, 1)) * innerH;
          return <circle key={i} cx={x} cy={y} r="3" fill="currentColor" className="text-shop" />;
        })}
      </svg>
      <div className="mt-2 flex items-center justify-between text-2xs text-muted-foreground">
        <span>Ú©Ù…ØªØ±ÛŒÙ†: {fmt(min)}</span>
        <span>Ù…ÛŒØ§Ù†Ú¯ÛŒÙ†: {fmt(data.averagePrice || 0)}</span>
        <span>Ø¨ÛŒØ´ØªØ±ÛŒÙ†: {fmt(max)}</span>

}

import { useGetReportSubjects, useSubmitShopReport } from '@/domains/front/shop/hooks/shop.hooks';
import { Select } from '@/components/primitives/Select/Select';
import { TextArea } from '@/components/primitives/TextArea/TextArea';
import { showToast } from '@/core/utils/toast';
import { toPersianDigits, getFullUrl } from '@/core/utils/formatters';

export interface SellersListProps {
  activeSellers: any;
  selectedSellerId: string | null;
  onSelectSeller: (id: string) => void;
  productTitle?: string;
}

const DESKTOP_MAX_VISIBLE = 4;

export function SellersList({
  activeSellers,
  selectedSellerId,
  onSelectSeller,
  productTitle = 'قطعه یدکی',
}: SellersListProps) {
  const [activeTab, setActiveTab] = React.useState<'online' | 'local'>('online');
  const [reportShop, setReportShop] = React.useState<any | null>(null);
  const [loadingSellerId, setLoadingSellerId] = React.useState<string | null>(null);
  const [isMapModalOpen, setIsMapModalOpen] = React.useState(false);
  const [mapInitialSellerId, setMapInitialSellerId] = React.useState<string | null>(null);
  const [isAllSellersModalOpen, setIsAllSellersModalOpen] = React.useState(false);

  const [selectedSubjectId, setSelectedSubjectId] = React.useState('');
  const [reportDescription, setReportDescription] = React.useState('');

  const addToBasket = useAddToBasket();
  const { data: reportSubjects = [] } = useGetReportSubjects('ShopProductReport');
  const submitReport = useSubmitShopReport();

  const onlineList = activeSellers.online || [];
  const localList = activeSellers.local || [];

  const visibleOnline = React.useMemo(() => {
    return onlineList.slice(0, DESKTOP_MAX_VISIBLE);
  }, [onlineList]);

  const hasMoreOnline = onlineList.length > DESKTOP_MAX_VISIBLE;

  const handleAddOnlineToBasket = async (e: React.MouseEvent, seller: any) => {
    e.preventDefault();
    e.stopPropagation();
    setLoadingSellerId(seller.id);
    try {
      await addToBasket.mutateAsync({ shopProductId: seller.id, quantity: 1 });
      showToast.success('قطعه با موفقیت به سبد خرید افزوده شد');
    } catch {} finally {
      setLoadingSellerId(null);
    }
  };

  const handleOpenMapWithSeller = (e: React.MouseEvent, seller: any) => {
    e.preventDefault();
    e.stopPropagation();
    setMapInitialSellerId(seller.id);
    setIsMapModalOpen(true);
  };

  const handleOpenGeneralMap = () => {
    setMapInitialSellerId(null);
    setIsMapModalOpen(true);
  };

  const handleReportSubmit = async () => {
    if (!selectedSubjectId) {
      showToast.error('لطفاً دلیل گزارش را انتخاب کنید');
      return;
    }
    try {
      await submitReport.mutateAsync({
        shopId: reportShop.shop.id,
        shopProductId: reportShop.id,
        reportSubjectId: selectedSubjectId,
        description: reportDescription,
      });
      showToast.success('گزارش با موفقیت ثبت شد و بررسی خواهد گردید');
      setReportShop(null);
      setSelectedSubjectId('');
      setReportDescription('');
    } catch (error: any) {
      showToast.error(error?.userMessage || 'خطا در ثبت گزارش');
    }
  };

  const renderSellerCard = (seller: any) => {
    const isOnline = activeTab === 'online';
    const isSelected = isOnline && selectedSellerId === seller.id;

    const originalPriceRaw = seller.retailPriceRaw || 0;
    const finalPriceRaw = seller.finalPriceRaw || 0;
    const calculatedPercent =
      originalPriceRaw > finalPriceRaw && originalPriceRaw > 0
        ? Math.round(((originalPriceRaw - finalPriceRaw) / originalPriceRaw) * 100)
        : 0;

    const discountPercent = seller.discountPercentage || calculatedPercent || 0;
    const hasDiscount = !!(
      (seller.hasDiscount || (originalPriceRaw > 0 && finalPriceRaw > 0 && originalPriceRaw > finalPriceRaw)) &&
      discountPercent > 0
    );

    return (
      <div
        key={seller.id}
        onClick={() => isOnline && onSelectSeller(seller.id)}
        className={cn(
          'relative flex w-full flex-col lg:flex-row lg:items-center lg:justify-between gap-3 lg:gap-4 rounded-2xl bg-card p-4 border shadow-xs transition-all duration-200',
          isSelected
            ? 'border-primary/50 bg-primary/5 ring-1 ring-primary/20'
            : 'border-border/80 hover:border-border hover:bg-muted/10',
          isOnline ? 'cursor-pointer' : 'cursor-default'
        )}
      >
        {hasDiscount && (
          <div className="flex lg:hidden items-center justify-between rounded-xl bg-destructive/10 px-3 py-1 text-2xs font-bold text-destructive">
            <div className="flex items-center gap-1">
              <Zap className="h-4 w-4 fill-destructive text-destructive" />
              <span>پیشنهاد ویژه</span>
            </div>
            <span className="font-bold">تخفیف {toPersianDigits(discountPercent)}٪</span>
          </div>
        )}

        <div className="flex items-center justify-between lg:justify-start gap-3 min-w-0 lg:w-[230px] xl:w-[270px] shrink-0">
          <div className="flex items-center gap-3 min-w-0 flex-1">
            <div className="flex h-10 w-10 shrink-0 items-center justify-center overflow-hidden rounded-xl border border-border/80 bg-muted/30 p-1">
              {seller.shop?.logo ? (
                <img src={getFullUrl(seller.shop.logo)} className="h-full w-full object-contain" alt="" />
              ) : (
                <Store className="h-5 w-5 text-muted-foreground" />
              )}
            </div>

            <div className="flex flex-col min-w-0 flex-1">
              <div className="flex items-center gap-1.5 flex-wrap">
                <span className="text-xs sm:text-sm font-bold text-foreground truncate max-w-[140px] sm:max-w-[180px]">
                  {seller.shop?.title}
                </span>
                <span className="rounded-md bg-muted px-2 py-0.5 text-3xs font-bold text-muted-foreground shrink-0">
                  {seller.typeLabel}
                </span>
              </div>

              <div className="mt-1 flex items-center gap-2 text-3xs text-muted-foreground">
                <span className="flex items-center gap-1 font-bold text-foreground">
                  <Star className="h-3 w-3 fill-warning text-warning" />
                  {toPersianDigits(seller.shop?.rate || '۴.۸')}
                </span>
                <span>•</span>
                <span className="flex items-center gap-1 truncate">
                  <MapPin className="h-3 w-3 text-muted-foreground shrink-0" />
                  <span className="truncate max-w-[90px]">
                    {(seller.shop as any)?.cityName || seller.shop?.address?.split('،')[0] || 'تهران'}
                  </span>
                </span>
              </div>
            </div>
          </div>

          <button
            type="button"
            title="گزارش خطای اطلاعات"
            onClick={(e) => {
              e.preventDefault();
              e.stopPropagation();
              setReportShop(seller);
            }}
            className="flex lg:hidden items-center gap-1 rounded-lg bg-muted px-2 py-1 text-3xs font-bold text-muted-foreground hover:bg-destructive/10 hover:text-destructive transition-colors shrink-0"
          >
            <AlertTriangle className="h-3 w-3" />
            <span>گزارش</span>
          </button>
        </div>

        <div className="flex flex-col gap-1.5 text-2xs text-muted-foreground lg:border-s lg:border-border/60 lg:ps-4 min-w-0 flex-1">
          <div className="flex items-center gap-2">
            <Truck className="h-4 w-4 text-primary shrink-0" />
            <span className="text-muted-foreground">ارسال:</span>
            <span className="font-bold text-foreground truncate">{seller.dayOfDeliveryLabel}</span>
          </div>
          <div className="flex items-center gap-2">
            <ShieldCheck className="h-4 w-4 text-emerald-600 dark:text-emerald-400 shrink-0" />
            <span className="font-bold text-emerald-600 dark:text-emerald-400 truncate">
              {seller.warrantyTitle || 'ضمانت اصالت و سلامت فیزیکی'}
            </span>
          </div>
        </div>

        <div className="flex items-center justify-between lg:justify-end gap-3 sm:gap-4 pt-3 lg:pt-0 border-t lg:border-t-0 border-dashed border-border/60 shrink-0 lg:ms-auto">
          <div className="flex flex-col items-start lg:items-end min-w-[90px] sm:min-w-[100px]">
            {hasDiscount && (
              <div className="flex items-center gap-1">
                <span className="text-3xs text-muted-foreground line-through">
                  {seller.retailPrice}
                </span>
                <span className="hidden sm:inline-flex rounded-full bg-destructive/10 text-destructive px-1.5 py-0.5 text-3xs font-bold">
                  {toPersianDigits(discountPercent)}٪
                </span>
              </div>
            )}
            <div className="flex items-baseline gap-1">
              <span className="text-sm sm:text-base font-black text-foreground" dir="ltr">
                {seller.finalPrice}
              </span>
              <span className="text-3xs text-muted-foreground">تومان</span>
            </div>
          </div>

          <div className="flex items-center gap-2 shrink-0">
            {isOnline ? (
              <Button
                variant="outline-primary"
                size="sm"
                onClick={(e) => handleAddOnlineToBasket(e, seller)}
                disabled={loadingSellerId === seller.id}
                className="h-9 gap-1.5 rounded-xl px-3 sm:px-4 text-xs font-bold shadow-xs whitespace-nowrap"
              >
                {loadingSellerId === seller.id ? (
                  <Loader2 aria-hidden="true" className="h-4 w-4 animate-spin" />
                ) : (
                  <>
                    <ShoppingBag aria-hidden="true" className="h-4 w-4" />
                    <span>افزودن به سبد</span>
                  </>
                )}
              </Button>
            ) : (
              <Button
                variant="outline"
                size="sm"
                onClick={(e) => handleOpenMapWithSeller(e, seller)}
                className="h-9 gap-1.5 rounded-xl px-3 sm:px-4 text-xs font-bold shadow-xs whitespace-nowrap"
              >
                <Phone aria-hidden="true" className="h-4 w-4" />
                <span>اطلاعات تماس</span>
              </Button>
            )}

            <button
              type="button"
              title="گزارش خطای اطلاعات"
              onClick={(e) => {
                e.preventDefault();
                e.stopPropagation();
                setReportShop(seller);
              }}
              className="hidden lg:flex h-9 w-9 items-center justify-center rounded-xl bg-muted/60 text-muted-foreground hover:bg-destructive/10 hover:text-destructive transition-colors shrink-0"
            >
              <AlertTriangle className="h-4 w-4" />
            </button>
          </div>
        </div>
      </div>
    );
  };

  return (
    <div className="relative mt-2 flex w-full select-none flex-col gap-4" dir="rtl">
      {/* تب‌های دوگانه چسبنده دقیقاً به زیر هدر در بالا (top-14 در موبایل و top-[68px] در دسکتاپ) */}
      <div className="sticky top-14 lg:top-[68px] z-20 flex w-full items-center border-b border-border bg-card/95 backdrop-blur-md shadow-xs">
        <button
          type="button"
          onClick={() => setActiveTab('online')}
          className={cn(
            'flex-1 border-b-2 py-3 text-center text-xs font-bold outline-none transition-all sm:flex-none sm:px-6 md:text-sm',
            activeTab === 'online'
              ? 'border-primary text-primary font-black'
              : 'border-transparent text-muted-foreground hover:text-foreground'
          )}
        >
          خرید اینترنتی ({toPersianDigits(onlineList.length)} فروشگاه)
        </button>
        <button
          type="button"
          onClick={() => setActiveTab('local')}
          className={cn(
            'flex-1 border-b-2 py-3 text-center text-xs font-bold outline-none transition-all sm:flex-none sm:px-6 md:text-sm',
            activeTab === 'local'
              ? 'border-primary text-primary font-black'
              : 'border-transparent text-muted-foreground hover:text-foreground'
          )}
        >
          خرید حضوری ({toPersianDigits(localList.length)} فروشگاه)
        </button>
      </div>

      {/* محتوای لیست فروشندگان */}
      <div className="flex flex-col gap-3">
        {activeTab === 'online' ? (
          onlineList.length > 0 ? (
            <>
              <div className="flex flex-col gap-3">
                {visibleOnline.map((s: any) => renderSellerCard(s))}
              </div>

              {hasMoreOnline && (
                <div className="flex w-full justify-center pt-2">
                  <Button
                    variant="outline"
                    onClick={() => setIsAllSellersModalOpen(true)}
                    className="h-10 gap-2 rounded-2xl px-6 text-xs font-bold border-primary/40 text-primary hover:bg-primary/5 shadow-2xs"
                  >
                    <Maximize2 className="h-4 w-4" />
                    <span>مشاهده همه فروشندگان ({toPersianDigits(onlineList.length)} فروشگاه)</span>
                  </Button>
                </div>
              )}
            </>
          ) : (
            <span className="py-8 text-center text-xs text-muted-foreground">
              فروشگاه اینترنتی فعالی برای این قطعه موجود نیست.
            </span>
          )
        ) : (
          <>
            {localList.length > 0 ? (
              <div className="flex flex-col gap-3">
                {localList.map((s: any) => renderSellerCard(s))}
              </div>
            ) : (
              <span className="py-8 text-center text-xs text-muted-foreground">
                فروشگاه حضوری فعالی برای این قطعه ثبت نشده است.
              </span>
            )}
            {localList.length > 0 && (
              <div
                onClick={handleOpenGeneralMap}
                className="group mt-2 flex h-20 w-full cursor-pointer select-none items-center justify-center gap-3 overflow-hidden rounded-2xl border border-border/80 bg-muted/20 p-4 text-center shadow-2xs transition-all hover:border-primary/40"
              >
                <div className="shrink-0 rounded-xl border border-primary/30 bg-primary/10 p-2 text-primary">
                  <MapPin aria-hidden="true" className="h-4 w-4" />
                </div>
                <span className="text-xs font-bold text-foreground">
                  مشاهده تمام فروشگاه‌های حضوری روی نقشه شهر
                </span>
              </div>
            )}
          </>
        )}
      </div>

      <Modal
        isOpen={isAllSellersModalOpen}
        onClose={() => setIsAllSellersModalOpen(false)}
        className="w-full max-w-4xl overflow-hidden rounded-2xl"
      >
        <ModalHeader onClose={() => setIsAllSellersModalOpen(false)}>
          <ModalTitle className="flex items-center gap-2 text-sm font-black text-foreground">
            <Store className="h-5 w-5 text-primary" />
            <span>لیست کامل فروشندگان این قطعه ({toPersianDigits(onlineList.length)} فروشگاه)</span>
          </ModalTitle>
        </ModalHeader>
        <ModalBody className="p-4 md:p-5 max-h-[80dvh] overflow-y-auto no-scrollbar flex flex-col gap-3" dir="rtl">
          {onlineList.map((s: any) => renderSellerCard(s))}
        </ModalBody>
      </Modal>

      <Modal isOpen={!!reportShop} onClose={() => setReportShop(null)} className="w-full max-w-md overflow-hidden rounded-2xl">
        <ModalHeader onClose={() => setReportShop(null)}>
          <ModalTitle className="flex items-center gap-2 text-sm font-bold text-foreground">
            <AlertTriangle aria-hidden="true" className="h-5 w-5 text-destructive" />
            <span>گزارش خطا در قیمت فروشگاه {reportShop?.shop?.title}</span>
          </ModalTitle>
        </ModalHeader>
        <ModalBody className="flex flex-col gap-4 p-5 pt-3" dir="rtl">
          <Select
            label="دلیل گزارش *"
            placeholder="دلیل گزارش را انتخاب کنید..."
            value={selectedSubjectId}
            onChange={(e) => setSelectedSubjectId(e.target.value)}
            options={reportSubjects.map((sub: any) => ({ value: sub.id, label: sub.title }))}
          />
          <TextArea
            label="توضیحات تکمیلی"
            placeholder="جزئیات خطا را بنویسید..."
            value={reportDescription}
            onChange={(e) => setReportDescription(e.target.value)}
            className="h-24 text-xs"
          />
          <div className="mt-2 flex w-full gap-3 border-t border-border/60 pt-4">
            <Button
              variant="outline"
              onClick={() => {
                setReportShop(null);
                setSelectedSubjectId('');
                setReportDescription('');
              }}
              className="h-10 flex-1 rounded-xl text-xs font-bold"
            >
              انصراف
            </Button>
            <Button
              variant="destructive"
              onClick={handleReportSubmit}
              isLoading={submitReport.isPending}
              className="h-10 flex-1 rounded-xl text-xs font-bold shadow-xs"
            >
              ثبت گزارش خطا
            </Button>
          </div>
        </ModalBody>
      </Modal>

      <InPersonSellersMapModal
        isOpen={isMapModalOpen}
        onClose={() => setIsMapModalOpen(false)}
        sellers={localList}
        productTitle={productTitle}
        initialActiveSellerId={mapInitialSellerId}
      />
    </div>
  );
}

'use client';

import * as React from 'react';
import { createPortal } from 'react-dom';
import { ChevronDown, ChevronUp } from 'lucide-react';
import { toPersianDigits } from '@/core/utils/formatters';
import { cn } from '@/design-system/utils/cn';

export interface SpecHighlightsProps {
  specGroups: any[];
  onSeeAll?: () => void;
}

function TruncatedSpec({ name, value }: { name: string; value: string }) {
  const [isOpen, setIsOpen] = React.useState(false);
  const [isTruncated, setIsTruncated] = React.useState(false);
  const [tooltipPos, setTooltipPos] = React.useState({ top: 0, left: 0 });
  const textRef = React.useRef<HTMLSpanElement>(null);
  const timerRef = React.useRef<NodeJS.Timeout | null>(null);
  const [mounted, setMounted] = React.useState(false);

  React.useEffect(() => {
    setMounted(true);
  }, []);

  React.useEffect(() => {
    const checkTruncation = () => {
      if (textRef.current) {
        const element = textRef.current;
        setIsTruncated(element.scrollWidth > element.clientWidth);
      }
    };
    checkTruncation();
    window.addEventListener('resize', checkTruncation);
    return () => window.removeEventListener('resize', checkTruncation);
  }, [value]);

  const updateTooltipPosition = () => {
    if (textRef.current) {
      const rect = textRef.current.getBoundingClientRect();
      setTooltipPos({ top: rect.bottom + 8, left: rect.left + rect.width / 2 });
    }
  };

  const handleMouseEnter = () => {
    if (!isTruncated) return;
    updateTooltipPosition();
    timerRef.current = setTimeout(() => setIsOpen(true), 250);
  };

  const handleMouseLeave = () => {
    if (timerRef.current) clearTimeout(timerRef.current);
    setIsOpen(false);
  };

  const handleClick = (e: React.MouseEvent) => {
    if (!isTruncated) return;
    e.stopPropagation();
    updateTooltipPosition();
    setIsOpen((prev) => !prev);
  };

  React.useEffect(() => {
    if (!isOpen) return;
    const handleClickOutside = () => setIsOpen(false);
    document.addEventListener('click', handleClickOutside);
    window.addEventListener('scroll', handleClickOutside, true);
    return () => {
      document.removeEventListener('click', handleClickOutside);
      window.removeEventListener('scroll', handleClickOutside, true);
    };
  }, [isOpen]);

  React.useEffect(() => {
    return () => {
      if (timerRef.current) clearTimeout(timerRef.current);
    };
  }, []);

  return (
    <div className="flex min-w-0 flex-col gap-1 rounded-md border border-border bg-muted/60 p-3">
      <span className="block truncate text-2xs font-bold text-muted-foreground md:text-xs">
        {toPersianDigits(name)}
      </span>

      <span
        ref={textRef}
        onClick={handleClick}
        onMouseEnter={handleMouseEnter}
        onMouseLeave={handleMouseLeave}
        className={cn(
          'mt-0.5 block truncate text-xs font-medium text-foreground md:text-sm',
          isTruncated && 'cursor-help'
        )}
      >
        {toPersianDigits(value)}
      </span>

      {mounted && isOpen && isTruncated
        ? createPortal(
            <div
              style={{
                position: 'fixed',
                top: tooltipPos.top,
                left: tooltipPos.left,
                transform: 'translateX(-50%)',
                zIndex: 9999,
              }}
              className="pointer-events-none"
              dir="rtl"
            >
              <div className="relative max-w-xs break-words rounded-md bg-foreground px-3 py-2 text-2xs font-bold text-background shadow-lg">
                {toPersianDigits(value)}
                <div className="absolute -top-1 left-1/2 h-2 w-2 -translate-x-1/2 rotate-45 bg-foreground" />
              </div>
            </div>,
            document.body
          )
        : null}
    </div>
  );
}

  const hasMoreThanFour = mainSpecs.length > 4;

  return (
    <div className="flex w-full select-none flex-col gap-2.5">
      <span className="me-1 text-sm font-bold text-foreground md:text-base">ÙˆÛŒÚ˜Ú¯ÛŒâ€ŒÙ‡Ø§ÛŒ Ù…Ù‡Ù… Ù…Ø­ØµÙˆÙ„</span>

      <div className="grid w-full grid-cols-1 gap-2.5 sm:grid-cols-2">

import { PickupAlert } from './components/PickupAlert';
import { SellersList } from './components/SellersList';
import { ProductSpecsTable } from './components/ProductSpecsTable';
import { ProductCommentsSection } from './components/ProductCommentsSection';
import { ProductInquiriesSection } from './components/ProductInquiriesSection';
import { ProductSideInvoice } from './components/ProductSideInvoice';
import { ProductPageSkeleton } from './components/ProductPageSkeleton';
import { ProductGallery } from './components/ProductGallery';
import { ProductMinimalHeader } from './components/ProductMinimalHeader';
import { MobileBottomAction } from '@/components/composites/MobileBottomAction/MobileBottomAction';
import { Modal, ModalHeader, ModalTitle, ModalBody, ModalFooter } from '@/components/composites/Modal/Modal';
import { Button } from '@/components/primitives/Button/Button';
import { Select } from '@/components/primitives/Select/Select';
import { TextArea } from '@/components/primitives/TextArea/TextArea';
import { toPersianDigits, getProductUrl } from '@/core/utils/formatters';
import { cn } from '@/design-system/utils/cn';
import { showToast } from '@/core/utils/toast';

interface ProductContentProps {
  productCode: number;
  initialData?: any;
}

export function ProductContent({ productCode, initialData }: ProductContentProps) {
  const { data: clientData, isLoading } = useGetProductPageData(productCode);
  const pageData = clientData || initialData;

  const { data: basketData } = useGetBasket();
  const addToBasket = useAddToBasket();
  const deleteFromBasket = useDeleteFromBasket();

  const defaultCondition = React.useMemo((): 'New' | 'Stock' | 'TakeOff' => {
    if (!pageData?.shopProducts) return 'New';
    const sp = pageData.shopProducts;
    const hasNew = (sp.newOnline?.length || 0) + (sp.newLocal?.length || 0) > 0 || !!sp.newNominated;
    if (hasNew) return 'New';
    const hasStock = (sp.stockOnline?.length || 0) + (sp.stockLocal?.length || 0) > 0 || !!sp.stockNominated;
    if (hasStock) return 'Stock';
    const hasTakeOff = (sp.takeOffOnline?.length || 0) + (sp.takeOffLocal?.length || 0) > 0 || !!sp.takeOffNominated;
    if (hasTakeOff) return 'TakeOff';
    return 'New';
  }, [pageData]);

  const [selectedCondition, setSelectedCondition] = React.useState<'New' | 'Stock' | 'TakeOff'>(defaultCondition);
  const [activeSellerId, setActiveSellerId] = React.useState<string | null>(null);
  const [isIntroExpanded, setIsIntroExpanded] = React.useState(false);
  const [activeSection, setActiveSection] = React.useState('intro');
  const [isMobileSubmitting, setIsMobileSubmitting] = React.useState(false);
  const [isBasketUpdating, setIsBasketUpdating] = React.useState(false);
  const [isNotifying, setIsNotifying] = React.useState(false);
  const [isNotifySuccessOpen, setIsNotifySuccessOpen] = React.useState(false);
  const [isReportModalOpen, setIsReportModalOpen] = React.useState(false);
  const [isPriceChartOpen, setIsPriceChartOpen] = React.useState(false);
  const [metaIndex, setMetaIndex] = React.useState(0);

  const [isSheetAttached, setIsSheetAttached] = React.useState(false);

  const [selectedSubjectId, setSelectedSubjectId] = React.useState('');
  const [reportDescription, setReportDescription] = React.useState('');
  const submitProductReport = useSubmitProductReport();
  const { data: reportSubjects = [] } = useGetReportSubjects('ProductReport');

  const mobileViewportRef = React.useRef<HTMLDivElement>(null);
  const sheetRef = React.useRef<HTMLDivElement>(null);

  const introRef = React.useRef<HTMLDivElement>(null);
  const specsRef = React.useRef<HTMLDivElement>(null);
  const commentsRef = React.useRef<HTMLDivElement>(null);
  const inquiriesRef = React.useRef<HTMLDivElement>(null);

  React.useEffect(() => {
    if (pageData && defaultCondition) setSelectedCondition(defaultCondition);
  }, [defaultCondition, pageData]);

  // اسکرول لیسنر دسکتاپ
  React.useEffect(() => {
    const handleWindowScroll = () => {
      if (window.innerWidth < 1024) return;
      const scrollPosition = window.scrollY + 140;
      const sections = [
        { id: 'intro', ref: introRef },
        { id: 'specs', ref: specsRef },
        { id: 'comments', ref: commentsRef },
        { id: 'inquiries', ref: inquiriesRef },
      ];
      for (const section of sections) {
        if (section.ref.current) {
          const top = section.ref.current.offsetTop;
          const height = section.ref.current.offsetHeight;
          if (scrollPosition >= top && scrollPosition < top + height) {
            setActiveSection(section.id);
            break;
          }
        }
      }
    };
    window.addEventListener('scroll', handleWindowScroll, { passive: true });
    return () => window.removeEventListener('scroll', handleWindowScroll);
  }, []);

  // اسکرول لیسنر پیوسته موبایل
  const handleMobileViewportScroll = () => {
    if (!mobileViewportRef.current) return;
    const scrollTop = mobileViewportRef.current.scrollTop;

    const attached = scrollTop >= 370;
    if (attached !== isSheetAttached) {
      setIsSheetAttached(attached);
    }

    const scrollPos = scrollTop + 120;
    const sections = [
      { id: 'intro', ref: introRef },
      { id: 'specs', ref: specsRef },
      { id: 'comments', ref: commentsRef },
      { id: 'inquiries', ref: inquiriesRef },
    ];
    for (const section of sections) {
      if (section.ref.current) {
        const top = section.ref.current.offsetTop;
        const height = section.ref.current.offsetHeight;
        if (scrollPos >= top && scrollPos < top + height) {
          setActiveSection(section.id);
          break;
        }
      }
    }
  };

  React.useEffect(() => {
    if (!pageData) return;
    const interval = setInterval(() => setMetaIndex((prev) => (prev + 1) % 3), 3000);
    return () => clearInterval(interval);
  }, [pageData]);

  if (!pageData && isLoading) return <ProductPageSkeleton />;
  if (!pageData) return null;

  const product = pageData.product;
  const sellersGroup = pageData.shopProducts || {};

  const getSellersForCondition = () => {
    if (selectedCondition === 'Stock') {
      return {
        nominated: sellersGroup.stockNominated || null,
        online: sellersGroup.stockOnline || [],
        local: sellersGroup.stockLocal || [],
      };
    }
    if (selectedCondition === 'TakeOff') {
      return {
        nominated: sellersGroup.takeOffNominated || null,
        online: sellersGroup.takeOffOnline || [],
        local: sellersGroup.takeOffLocal || [],
      };
    }
    return {
      nominated: sellersGroup.newNominated || null,
      online: sellersGroup.newOnline || [],
      local: sellersGroup.newLocal || [],
    };
  };

  const activeSellers = getSellersForCondition();
  const allAvailableSellersList = [...(activeSellers.online || []), ...(activeSellers.local || [])];

  const currentSelectedSeller =
    allAvailableSellersList.find((s) => s.id === activeSellerId) ||
    activeSellers.nominated ||
    allAvailableSellersList[0] ||
    null;

  const basketItems: any[] = React.useMemo(() => {
    if (!basketData) return [];
    if (Array.isArray(basketData.items)) return basketData.items;
    if (Array.isArray((basketData as any).subBaskets)) {
      return (basketData as any).subBaskets.flatMap((sb: any) => sb.items || sb.subBasketItems || []);
    }
    return [];
  }, [basketData]);

  const currentBasketItem = React.useMemo(() => {
    if (!currentSelectedSeller || basketItems.length === 0) return null;
    return (
      basketItems.find(
        (it: any) => (it.shopProductId || it.shopProduct?.id) === currentSelectedSeller.id
      ) || null
    );
  }, [currentSelectedSeller, basketItems]);

  const isInBasket = !!currentBasketItem && currentBasketItem.quantity > 0;
  const basketQuantity = currentBasketItem?.quantity || 0;

  const maxAllowedQuantity = React.useMemo(() => {
    const stockQuantity = Number(currentSelectedSeller?.quantity);
    if (!isNaN(stockQuantity) && stockQuantity > 0) return stockQuantity;
    return Number(
      currentBasketItem?.inventory?.maxQuantity ||
      currentBasketItem?.maxQuantity ||
      currentSelectedSeller?.maxQuantityPerOrder ||
      10
    );
  }, [currentSelectedSeller, currentBasketItem]);

  const galleryImages = React.useMemo(() => {
    const all = [product.image, ...(product.gallery || [])].filter(Boolean) as string[];
    return Array.from(new Set(all));
  }, [product.image, product.gallery]);

  const handleScrollToSection = (ref: React.RefObject<HTMLDivElement | null>, sectionId: string) => {
    if (window.innerWidth < 1024) {
      if (mobileViewportRef.current && ref.current) {
        const topOffset = ref.current.offsetTop - 56;
        mobileViewportRef.current.scrollTo({ top: topOffset, behavior: 'smooth' });
        setActiveSection(sectionId);
      }
    } else {
      if (ref.current) {
        const topOffset = ref.current.getBoundingClientRect().top + window.scrollY - 120;
        window.scrollTo({ top: topOffset, behavior: 'smooth' });
        setActiveSection(sectionId);
      }
    }
  };

  const handleMobileAddToBasket = async () => {
    if (!currentSelectedSeller) return;
    setIsMobileSubmitting(true);
    try {
      await addToBasket.mutateAsync({ shopProductId: currentSelectedSeller.id, quantity: 1 });
      showToast.success('قطعه با موفقیت به سبد خرید شما افزوده شد');
    } catch (err: any) {
      showToast.error(err?.userMessage || 'خطا در افزودن به سبد خرید');
    } finally {
      setIsMobileSubmitting(false);
    }
  };

  const handleIncreaseBasket = async () => {
    if (!currentSelectedSeller || isBasketUpdating) return;
    if (basketQuantity >= maxAllowedQuantity) {
      showToast.error(`حداکثر موجودی قابل سفارش ${toPersianDigits(maxAllowedQuantity)} عدد است`);
      return;
    }
    setIsBasketUpdating(true);
    try {
      await addToBasket.mutateAsync({ shopProductId: currentSelectedSeller.id, quantity: 1 });
      showToast.success('تعداد کالا افزایش یافت');
    } catch (err: any) {
      showToast.error(err?.userMessage || 'خطا در افزایش تعداد');
    } finally {
      setIsBasketUpdating(false);
    }
  };

  const handleDecreaseBasket = async () => {
    if (!currentSelectedSeller || isBasketUpdating || basketQuantity <= 0) return;
    setIsBasketUpdating(true);
    try {
      await deleteFromBasket.mutateAsync({ shopProductId: currentSelectedSeller.id, quantity: 1 });
      if (basketQuantity === 1) {
        showToast.success('کالا از سبد خرید حذف شد');
      } else {
        showToast.success('تعداد کالا کاهش یافت');
      }
    } catch (err: any) {
      showToast.error(err?.userMessage || 'خطا در کاهش تعداد');
    } finally {
      setIsBasketUpdating(false);
    }
  };

  const handleNotifyMe = async () => {
    if (isNotifying) return;
    setIsNotifying(true);
    try {
      await new Promise((r) => setTimeout(r, 600));
      setIsNotifySuccessOpen(true);
    } finally {
      setIsNotifying(false);
    }
  };

  const handleReportSubmit = async () => {
    if (!selectedSubjectId) {
      showToast.error('لطفاً دلیل گزارش کالا را انتخاب کنید');
      return;
    }
    try {
      await submitProductReport.mutateAsync({
        productId: product.id,
        reportSubjectId: selectedSubjectId,
        description: reportDescription,
      });
      showToast.success('گزارش خطای کالا با موفقیت ثبت شد');
      setIsReportModalOpen(false);
      setSelectedSubjectId('');
      setReportDescription('');
    } catch (error: any) {
      showToast.error(error?.userMessage || 'خطا در ثبت گزارش');
    }
  };

  const originalPriceRaw = currentSelectedSeller?.retailPriceRaw || 0;
  const finalPriceRaw = currentSelectedSeller?.finalPriceRaw || 0;
  const calculatedPercent =
    originalPriceRaw > finalPriceRaw
      ? Math.round(((originalPriceRaw - finalPriceRaw) / originalPriceRaw) * 100)
      : 0;

  const discountPercentage = currentSelectedSeller?.discountPercentage || calculatedPercent || 0;
  const hasDiscount = !!(currentSelectedSeller?.hasDiscount && discountPercentage > 0);
  const discountExpiration = (currentSelectedSeller as any)?.discountUntil || null;
  const hasTimedDiscount = hasDiscount && !!discountExpiration;

  const metaItems = [
    `فروشگاه: ${currentSelectedSeller?.shop?.title || 'یدکچی'}`,
    `${toPersianDigits(product.views || 0)} بازدید اخیر`,
    `${toPersianDigits(product.salesCount || 0)} خرید موفق`,
  ];

  const leftPriceContent = currentSelectedSeller ? (
    <div className="flex h-10 w-36 max-w-[150px] select-none flex-col justify-between">
      <div className="relative h-4 w-full overflow-hidden">
        <AnimatePresence mode="wait">
          <motion.span
            key={metaIndex}
            initial={{ y: 10, opacity: 0 }}
            animate={{ y: 0, opacity: 1 }}
            exit={{ y: -10, opacity: 0 }}
            transition={{ duration: 0.2 }}
            className="absolute block w-full truncate text-2xs font-bold leading-none text-muted-foreground"
          >
            {metaItems[metaIndex]}
          </motion.span>
        </AnimatePresence>
      </div>

      <div className="flex flex-col">
        {currentSelectedSeller.hasDiscount ? (
          <span className="mb-1 text-2xs leading-none text-muted-foreground line-through">
            {currentSelectedSeller.retailPrice}
          </span>
        ) : null}
        <span className="text-xs font-bold leading-none text-foreground md:text-sm">
          {currentSelectedSeller.finalPrice}
        </span>
      </div>
    </div>
  ) : null;

  // رندر بردکرامب
  const renderBreadcrumb = () => (
    <div className="flex w-full select-none items-center gap-2 overflow-x-auto py-2 text-2xs text-muted-foreground scrollbar-none md:text-xs">
      <Link href="/" className="transition-colors hover:text-primary whitespace-nowrap shrink-0">
        خانه
      </Link>
      {product.breadCrumbs?.map((crumb: any, idx: number) => {
        const isLast = idx === product.breadCrumbs.length - 1;
        const url = isLast
          ? getProductUrl(product.code, product.title)
          : `/part-category/${crumb.englishTitle}`;
        return (
          <div key={crumb.id || idx} className="flex items-center gap-2 shrink-0">
            <ChevronLeft aria-hidden="true" className="h-4 w-4 shrink-0 text-muted-foreground" />
            {isLast ? (
              <span className="max-w-[170px] truncate font-bold text-foreground sm:max-w-none">
                {crumb.title}
              </span>
            ) : (
              <Link href={url} className="whitespace-nowrap transition-colors hover:text-primary">
                {crumb.title}
              </Link>
            )}
          </div>
        );
      })}
    </div>
  );

  return (
    <>
      {/* ======================================================== */}
      {/*        نسخه دسکتاپ (دقیقاً مشابه ساختار اول و بدون فضای خالی)  */}
      {/* ======================================================== */}
      <div className="hidden lg:flex w-full flex-col gap-1 mt-3" dir="rtl">
        {renderBreadcrumb()}

        <div className="flex w-full items-start gap-8">
          <div className="flex w-full min-w-0 flex-1 flex-col gap-6">
            {/* ردیف بالا: گالری و مشخصات روبروی هم */}
            <div className="flex w-full gap-8">
              <div className="w-[45%] shrink-0">
                <ProductGallery
                  images={galleryImages}
                  title={product.title}
                  hasDiscount={hasDiscount}
                  discountPercent={discountPercentage}
                  hasTimedDiscount={hasTimedDiscount}
                  discountExpiration={discountExpiration}
                />
              </div>

              <div className="flex min-w-0 flex-1 flex-col gap-5">
                <ProductHeader
                  product={product}
                  onScrollToComments={() => handleScrollToSection(commentsRef, 'comments')}
                  onScrollToInquiries={() => handleScrollToSection(inquiriesRef, 'inquiries')}
                />

                <ConditionSelector
                  selectedCondition={selectedCondition}
                  onChangeCondition={(cond) => {
                    setSelectedCondition(cond);
                    setActiveSellerId(null);
                  }}
                  sellersGroup={sellersGroup}
                />

                <SpecHighlights
                  specGroups={product.specGroups}
                  onSeeAll={() => handleScrollToSection(specsRef, 'specs')}
                />

                <PickupAlert />
              </div>
            </div>

            {/* ردیف پایین: در امتداد کل عرض (زیر گالری و اطلاعات) */}
            <SellersList
              activeSellers={activeSellers}
              selectedSellerId={currentSelectedSeller?.id || null}
              onSelectSeller={(id) => setActiveSellerId(id)}
              productTitle={product.title}
            />

            <div className="sticky top-[68px] z-30 flex h-12 w-full items-center gap-6 overflow-x-auto border-b border-border bg-card/95 backdrop-blur-md scrollbar-none">
              {[
                { id: 'intro', label: 'معرفی کالا', ref: introRef },
                { id: 'specs', label: 'مشخصات فنی', ref: specsRef },
                { id: 'comments', label: 'امتیاز و نظرات', ref: commentsRef },
                { id: 'inquiries', label: 'پرسش و پاسخ', ref: inquiriesRef },
              ].map((tab) => (
                <button
                  key={tab.id}
                  type="button"
                  onClick={() => handleScrollToSection(tab.ref, tab.id)}
                  className={cn(
                    'h-full shrink-0 border-b-2 px-1 text-xs font-bold transition-all flex items-center md:text-sm',
                    activeSection === tab.id
                      ? 'border-primary text-primary font-black'
                      : 'border-transparent text-muted-foreground hover:text-foreground'
                  )}
                >
                  {tab.label}
                </button>
              ))}
            </div>

            <div ref={introRef} className="border-b border-border pb-6 pt-6">
              <h3 className="mb-3 text-sm font-bold text-foreground md:text-base">معرفی کالا</h3>
              <div className="relative">
                <div
                  style={{ maxHeight: isIntroExpanded ? 'none' : '110px' }}
                  className="overflow-hidden text-justify text-xs leading-reading text-muted-foreground transition-all md:text-sm"
                  dangerouslySetInnerHTML={{
                    __html: product.description || 'توضیحی برای این کالا ثبت نشده است.',
                  }}
                />
                {!isIntroExpanded && (
                  <div className="pointer-events-none absolute inset-x-0 bottom-0 h-14 bg-gradient-to-t from-card to-transparent" />
                )}
              </div>
              <div className="mt-3 flex justify-start">
                <button
                  type="button"
                  onClick={() => setIsIntroExpanded(!isIntroExpanded)}
                  className="flex items-center gap-1 text-xs font-bold text-shop hover:underline"
                >
                  <span>{isIntroExpanded ? 'مشاهده کمتر' : 'مشاهده بیشتر'}</span>
                  {isIntroExpanded ? <ChevronUp className="h-4 w-4" /> : <ChevronDown className="h-4 w-4" />}
                </button>
              </div>
            </div>

            <div ref={specsRef} className="border-b border-border pb-6 pt-8">
              <ProductSpecsTable specGroups={product.specGroups} />
            </div>

            <div ref={commentsRef} className="border-b border-border pb-6 pt-8">
              <ProductCommentsSection
                productId={product.id}
                productTitle={product.title}
                productImage={product.image}
              />
            </div>

            <div ref={inquiriesRef} className="pt-8">
              <ProductInquiriesSection
                productId={product.id}
                productTitle={product.title}
                productImage={product.image}
              />
            </div>
          </div>

          {/* فاکتور کناری دسکتاپ */}
          <div className="w-[320px] shrink-0 sticky top-[132px] flex flex-col gap-6 xl:w-[340px]">
            <ProductSideInvoice
              product={product}
              seller={currentSelectedSeller}
              sellersCount={allAvailableSellersList.length}
            />
          </div>
        </div>
      </div>

      {/* ======================================================== */}
      {/*     نسخه موبایل (مودال شیت تعاملی با کنترل لمسی دو محوره)   */}
      {/* ======================================================== */}
      <div className="lg:hidden relative h-[100dvh] w-full overflow-hidden bg-background" dir="rtl">
        <ProductMinimalHeader
          productTitle={product.title}
          isAttached={isSheetAttached}
          onOpenReportModal={() => setIsReportModalOpen(true)}
          onOpenPriceChart={() => setIsPriceChartOpen(true)}
        />

        {/* لایه ثابت گالری با بردکرامب در بالای آن */}
        <div className="fixed top-14 inset-x-0 z-0 flex flex-col items-center bg-card select-none pointer-events-auto">
          {/* بردکرامب موبایل */}
          <div className="w-full px-4 border-b border-border/40 bg-card/60">
            {renderBreadcrumb()}
          </div>

          <div className="w-full max-w-2xl px-2">
            <ProductGallery
              images={galleryImages}
              title={product.title}
              hasDiscount={hasDiscount}
              discountPercent={discountPercentage}
              hasTimedDiscount={hasTimedDiscount}
              discountExpiration={discountExpiration}
              onDragVertical={(deltaY) => {
                if (mobileViewportRef.current) {
                  mobileViewportRef.current.scrollTop -= deltaY;
                }
              }}
            />
          </div>
        </div>

        {/* کانتینر اسکرول با تاچ عبوری در فضای خالی و ارتفاع بهینه شیت */}
        <div
          ref={mobileViewportRef}
          onScroll={handleMobileViewportScroll}
          className="relative z-10 h-full w-full overflow-y-auto overflow-x-hidden scrollbar-none overscroll-y-contain pt-[420px] sm:pt-[470px] pointer-events-none"
          style={{ WebkitOverflowScrolling: 'touch' }}
        >
          {/* شیت با گوشه‌های گرد و الحاق نرم به هدر */}
          <div
            ref={sheetRef}
            className={cn(
              'pointer-events-auto relative z-20 w-full min-h-screen bg-card border-t transition-all duration-300',
              isSheetAttached
                ? 'rounded-t-none border-border shadow-none'
                : 'rounded-t-[32px] border-border/80 shadow-[0_-16px_40px_rgba(0,0,0,0.16)] dark:shadow-[0_-16px_40px_rgba(0,0,0,0.5)]'
            )}
          >
            {/* دستگیره شیت */}
            <div
              onClick={() => {
                if (mobileViewportRef.current) {
                  mobileViewportRef.current.scrollTo({
                    top: isSheetAttached ? 0 : 400,
                    behavior: 'smooth',
                  });
                }
              }}
              className="flex h-7 w-full cursor-pointer items-center justify-center pt-2 select-none"
            >
              <div
                className={cn(
                  'h-1.5 w-12 rounded-full transition-opacity duration-300',
                  isSheetAttached ? 'opacity-0' : 'bg-muted-foreground/30 hover:bg-muted-foreground/50'
                )}
              />
            </div>

            {/* محتوای درون شیت در موبایل */}
            <div className="flex w-full flex-col gap-6 px-4 pb-32 pt-1">
              <ProductHeader
                product={product}
                onScrollToComments={() => handleScrollToSection(commentsRef, 'comments')}
                onScrollToInquiries={() => handleScrollToSection(inquiriesRef, 'inquiries')}
              />

              <ConditionSelector
                selectedCondition={selectedCondition}
                onChangeCondition={(cond) => {
                  setSelectedCondition(cond);
                  setActiveSellerId(null);
                }}
                sellersGroup={sellersGroup}
              />

              <SpecHighlights
                specGroups={product.specGroups}
                onSeeAll={() => handleScrollToSection(specsRef, 'specs')}
              />

              <PickupAlert />

              <SellersList
                activeSellers={activeSellers}
                selectedSellerId={currentSelectedSeller?.id || null}
                onSelectSeller={(id) => setActiveSellerId(id)}
                productTitle={product.title}
              />

              {/* تب‌های استیکی مماس به زیر هدر */}
              <div className="sticky top-14 z-30 flex h-12 w-full items-center gap-6 overflow-x-auto border-b border-border bg-card/95 backdrop-blur-md scrollbar-none">
                {[
                  { id: 'intro', label: 'معرفی کالا', ref: introRef },
                  { id: 'specs', label: 'مشخصات فنی', ref: specsRef },
                  { id: 'comments', label: 'امتیاز و نظرات', ref: commentsRef },
                  { id: 'inquiries', label: 'پرسش و پاسخ', ref: inquiriesRef },
                ].map((tab) => (
                  <button
                    key={tab.id}
                    type="button"

'use client';

import * as React from 'react';
import Link from 'next/link';
import Image from 'next/image';
import { Star, Hourglass } from 'lucide-react';
import { cn } from '@/design-system/utils/cn';
import { getProductUrl, toPersianDigits, getFullUrl } from '@/core/utils/formatters';
import { useImpression } from '@/shared/hooks/useImpression';

export interface ProductDealCardProps {
  product: any;
  serverTime?: string | Date;
  showTimer?: boolean;
  showRating?: boolean;
  className?: string;
}

export function ProductDealCard({
  product,
  serverTime,
  showTimer = true,
  showRating = true,
  className,
}: ProductDealCardProps) {
  const nominated = product?.nominatedShopProduct || {};

  const title =
    product?.productTitle ||
    product?.title ||
    product?.name?.value ||
    product?.name ||
    '';

  const originalPriceRaw =
    product?.rialRetailPrice ??
    product?.price?.raw ??
    product?.price ??
    nominated?.rialRetailPrice ??
    nominated?.price ??
    0;

  const finalPriceRaw =
    product?.rialFinalPrice ??
    product?.discountPrice ??
    nominated?.rialFinalPrice ??
    nominated?.discountPrice ??
    originalPriceRaw;

  const originalPriceToman = Math.round(originalPriceRaw / 10);
  const finalPriceToman = Math.round(finalPriceRaw / 10);

  const hasDiscount =
    product?.isDiscountApplied !== undefined
      ? product.isDiscountApplied
      : originalPriceRaw > finalPriceRaw && finalPriceRaw > 0;

  const discountPercent =
    product?.discountPercentage ??
    product?.discount?.percent ??
    nominated?.discountPercentage ??
    (hasDiscount && originalPriceRaw > 0
      ? Math.round(((originalPriceRaw - finalPriceRaw) / originalPriceRaw) * 100)
      : 0);

  const expirationStr =
    nominated.discountUntil ||
    nominated.discountExpiration ||
    product?.discountExpiration;
  const hasExpiration = Boolean(expirationStr);

  const [timeLeft, setTimeLeft] = React.useState({ hours: 0, minutes: 0, seconds: 0 });
  const [isDragging, setIsDragging] = React.useState(false);
  const dragStart = React.useRef({ x: 0, y: 0 });

  const shopProductId = product?.shopProductId || nominated?.id || product?.id || null;
  const impressionRef = useImpression(shopProductId);

  React.useEffect(() => {
    if (!hasExpiration) return;

    const calculateTime = () => {
      const serverDate = serverTime ? new Date(serverTime) : new Date();
      const clientDate = new Date();
      const timeOffset = serverDate.getTime() - clientDate.getTime();
      const syncedNow = new Date(Date.now() + timeOffset);
      const diff = +new Date(expirationStr as string) - +syncedNow;

      if (diff > 0) {
        setTimeLeft({
          hours: Math.floor(diff / (1000 * 60 * 60)),
          minutes: Math.floor((diff / 1000 / 60) % 60),
          seconds: Math.floor((diff / 1000) % 60),
        });
      } else {
        setTimeLeft({ hours: 0, minutes: 0, seconds: 0 });
      }
    };

    calculateTime();
    const interval = setInterval(calculateTime, 1000);
    return () => clearInterval(interval);
  }, [expirationStr, serverTime, hasExpiration]);

  const formatPersianDigits = (num: number) =>
    new Intl.NumberFormat('fa-IR', { useGrouping: false })
      .format(num)
      .padStart(2, '۰');

  const formatPrice = (value: number) => new Intl.NumberFormat('fa-IR').format(value);

  const handleMouseDown = (e: React.MouseEvent) => {
    dragStart.current = { x: e.clientX, y: e.clientY };
    setIsDragging(false);
  };

  const handleMouseMove = (e: React.MouseEvent) => {
    const dx = Math.abs(e.clientX - dragStart.current.x);
    const dy = Math.abs(e.clientY - dragStart.current.y);
    if (dx > 6 || dy > 6) setIsDragging(true);
  };

  const handleTouchStart = (e: React.TouchEvent) => {
    const touch = e.touches[0];
    if (touch) {
      dragStart.current = { x: touch.clientX, y: touch.clientY };
      setIsDragging(false);
    }
  };

  const handleTouchMove = (e: React.TouchEvent) => {
    const touch = e.touches[0];
    if (touch) {
      const dx = Math.abs(touch.clientX - dragStart.current.x);
      const dy = Math.abs(touch.clientY - dragStart.current.y);
      if (dx > 6 || dy > 6) setIsDragging(true);
    }
  };

  const handleClick = (e: React.MouseEvent) => {
    if (isDragging) {
      e.preventDefault();
      e.stopPropagation();
    }
  };

  const renderStars = () => {
    const rating = product?.averageRate || product?.rating?.average || 5;
    return (
      <div className="flex select-none items-center gap-1">
        <Star aria-hidden="true" className="h-3 w-3 shrink-0 fill-warning text-warning" />
        <span className="text-2xs font-medium text-muted-foreground sm:text-xs">
          امتیاز {toPersianDigits(rating)}
        </span>
      </div>
    );
  };

  const codeOrId =
    product?.productCode || product?.code || product?.productId || product?.id;
  const productCardUrl = getProductUrl(codeOrId, title);

  const imageSource =
    product?.image ||
    product?.images?.[0]?.medium ||
    product?.images?.[0]?.url ||
    product?.images?.[0] ||
    null;

  return (
    <Link
      href={productCardUrl}
      prefetch={false}
      className="block h-full w-full select-none"
      draggable={false}
      onMouseDown={handleMouseDown}
      onMouseMove={handleMouseMove}
      onTouchStart={handleTouchStart}
      onTouchMove={handleTouchMove}
      onClick={handleClick}
    >
      <div
        ref={impressionRef}
        className={cn(
          'relative flex h-full w-full select-none flex-col items-center rounded-lg border border-border bg-card p-3 transition-all duration-normal sm:p-3.5',
          'hover:border-primary/40 hover:shadow-sm',
          className
        )}
      >
        <div className="relative mb-2 aspect-[4/3] w-full select-none overflow-hidden rounded-md" draggable={false}>
          <Image
            src={getFullUrl(imageSource)}
            alt={product?.imageAlt || title}
            fill
            sizes="(max-width: 640px) 50vw, (max-width: 1024px) 33vw, 250px"
            className="select-none rounded-md object-contain"
            draggable={false}
          />
        </div>

        <div className="mb-1 flex min-h-[2.6rem] w-full items-start justify-end select-none">
          <h4 className="line-clamp-2 w-full text-sm font-semibold leading-relaxed text-foreground">
            {title}
          </h4>
        </div>

        {showRating ? (
          <div className="mb-3 mt-1.5 flex w-full select-none justify-start">
            {renderStars()}
          </div>
        ) : null}

        <div className="mt-auto flex w-full items-center justify-between pt-2">
          <div
            className={cn(
              'shrink-0 rounded-sm border border-accent bg-accent px-2.5 py-1 text-xs font-bold text-accent-foreground transition-opacity duration-fast',
              hasDiscount && discountPercent > 0 ? 'opacity-100' : 'pointer-events-none opacity-0'
            )}
          >
            {toPersianDigits(discountPercent)}٪
          </div>

          <div className="flex min-w-0 flex-col items-end">
            {hasDiscount && originalPriceToman > finalPriceToman ? (
              <span className="text-2xs font-medium text-muted-foreground line-through sm:text-xs">
                {formatPrice(originalPriceToman)}
              </span>
            ) : null}
            <div className="mt-0.5 flex items-center gap-0.5">
              <span className="text-base font-bold text-foreground sm:text-lg">
                {formatPrice(finalPriceToman)}
              </span>
              <span className="text-2xs text-muted-foreground">تومان</span>
            </div>
          </div>
        </div>

        {showTimer && hasExpiration ? (
          <div className="mt-2.5 flex w-full items-center justify-between border-t border-dashed border-border pt-1.5 text-muted-foreground">
            <div className="flex items-center gap-1 font-bold text-foreground" dir="ltr">
              <span className="rounded-sm bg-muted px-1.5 py-0.5 text-2xs">
                {formatPersianDigits(timeLeft.hours)}
              </span>
              <span className="text-muted-foreground">:</span>
              <span className="rounded-sm bg-muted px-1.5 py-0.5 text-2xs">

  const [tickerIndex, setTickerIndex] = React.useState(0);
  const [isMounted, setIsMounted] = React.useState(false);
  const [activeLoading, setActiveLoading] = React.useState(false);

  const { data: rawBasket } = useGetBasket();
  const addToBasket = useAddToBasket();
  const deleteFromBasket = useDeleteFromBasket();

  const rawBasketData = rawBasket as any;
  const lastValidBasketRef = React.useRef<any>(null);

  if (rawBasketData && !rawBasketData.isEmpty) {
    lastValidBasketRef.current = rawBasketData;
  }
  const basket =
    rawBasketData && !rawBasketData.isEmpty ? rawBasketData : lastValidBasketRef.current;

  React.useEffect(() => {
    setIsMounted(true);
  }, []);

  const nominated = product?.nominatedShopProduct || {};
  const shopProductId = product?.shopProductId || nominated?.id || product?.id || null;
  const impressionRef = useImpression(shopProductId);

  const title =
    product?.productTitle || product?.title || product?.name?.value || product?.name || '';

  // استخراج فیلدهای درخواستی شما برای اسلایدر
  const shopName = product?.shopTitle || nominated?.shopTitle || product?.shop?.name || '';
  const scoreValue = product?.score ?? nominated?.score ?? product?.averageRate ?? 5;
  const quantityValue = product?.quantity ?? nominated?.quantity ?? 0;

  const originalPriceRaw = Number(
    product?.rialRetailPrice ??
      nominated?.rialRetailPrice ??
      product?.price?.raw ??
      (typeof product?.price === 'number' ? product.price : 0) ??
      nominated?.price ??
      0
  );

  const finalPriceRaw = Number(
    product?.rialFinalPrice ??
      nominated?.rialFinalPrice ??
      (product?.discount?.hasDiscount
        ? product?.price?.raw * (1 - product?.discount?.percent / 100)
        : null) ??
      product?.price?.raw ??
      (typeof product?.price === 'number' ? product.price : 0) ??
      originalPriceRaw
  );

  const originalPriceToman = Math.round(originalPriceRaw / 10);
  const finalPriceToman = Math.round(finalPriceRaw / 10);

  const isOutOfStock =
    finalPriceRaw === 0 || (product?.quantity !== undefined && product.quantity === 0);

  const hasDiscount =
    product?.isDiscountApplied !== undefined
      ? product.isDiscountApplied
      : originalPriceRaw > finalPriceRaw && !isOutOfStock;

  const discountPercent =
    product?.discountPercentage ??
    (hasDiscount && originalPriceRaw > 0
      ? Math.round(((originalPriceRaw - finalPriceRaw) / originalPriceRaw) * 100)
      : nominated?.discountPercentage || product?.discount?.percent || 0);

  const basketItem = React.useMemo(() => {
    if (!basket || basket.isEmpty || !shopProductId) return null;
    for (const sub of basket.subBaskets) {
      const found = sub.items.find((item: any) => item.shopProductId === shopProductId);
      if (found) return found;
    }
    return null;
  }, [basket, shopProductId]);

  const isInBasket = Boolean(basketItem);
  const basketQuantity = basketItem?.quantity || 0;
  const maxLimit =
    basketItem?.maxQuantity ||
    product?.quantity ||
    nominated?.maxQuantityPerOrder ||
    nominated?.quantity ||
    10;

  // ساخت دقیق اسلایدر فقط بین این 3 مورد: shopTitle, score, quantity
  const tickerItems = React.useMemo(() => {
    const items: { text: string; icon: any; isShop?: boolean }[] = [];

    // ۱. نام فروشگاه (اگر خالی نبود یا خواستید همیشه باشه میتونید شرط رو بردارید)
    items.push({
      text: shopName ? `فروشگاه: ${shopName}` : 'فروشگاه: یدکچی',
      icon: Store,
      isShop: true,
    });

    // ۲. امتیاز (Score)
    items.push({
      text: `امتیاز: ${toPersianDigits(scoreValue)} از ۵`,
      icon: Star,
    });

    // ۳. موجودی (Quantity)
    items.push({
      text: quantityValue > 0 ? `موجودی در انبار: ${toPersianDigits(quantityValue)} عدد` : 'وضعیت: ناموجود در انبار',
      icon: Package,
    });

    return items;
  }, [shopName, scoreValue, quantityValue]);

  const tickerLength = tickerItems.length;

  // اسلایدر خودکار هر ۳ ثانیه
  React.useEffect(() => {
    if (!isMounted || tickerLength <= 1) return;
    const interval = setInterval(() => {
      setTickerIndex((prev) => (prev + 1) % tickerLength);
    }, 3000);
    return () => clearInterval(interval);
  }, [tickerLength, isMounted]);

  const formatPrice = (value: number) => new Intl.NumberFormat('fa-IR').format(value);

  const handleAddToBasket = async (e: React.MouseEvent) => {
    e.preventDefault();
    e.stopPropagation();
    if (!shopProductId || isOutOfStock) return;
    setActiveLoading(true);
    try {
      await addToBasket.mutateAsync({ shopProductId, quantity: 1 });
      showToast.success('قطعه به سبد خرید اضافه شد');
    } catch (err: any) {
    } finally {
      setActiveLoading(false);
    }
  };

  const handleIncrease = async (e: React.MouseEvent) => {
    e.preventDefault();
    e.stopPropagation();
    if (!shopProductId || activeLoading || basketQuantity >= maxLimit) return;
    setActiveLoading(true);
    try {
      await addToBasket.mutateAsync({ shopProductId, quantity: 1 });
      showToast.success('تعداد کالا افزایش یافت');
    } catch (err: any) {
    } finally {
      setActiveLoading(false);
    }
  };

  const handleDecrease = async (e: React.MouseEvent) => {
    e.preventDefault();
    e.stopPropagation();
    if (!shopProductId || activeLoading) return;
    setActiveLoading(true);
    try {
      await deleteFromBasket.mutateAsync({ shopProductId, quantity: 1 });
      showToast.success('تعداد کالا کاهش یافت');
    } catch (err: any) {
    } finally {
      setActiveLoading(false);
    }
  };

  const handleCardClick = () => {
    if (shopProductId) trackShopProductClick(shopProductId);
  };

  const ratingValue = product?.averageRate || product?.rating?.average || scoreValue;
  const ratingCount = product?.rateCount || product?.rating?.count || 0;

  const CurrentTickerIcon = tickerItems[tickerIndex]?.icon || Store;
  const currentItem = tickerItems[tickerIndex];

  const codeOrId =
    product?.productCode || product?.code || product?.productId || product?.id;
  const productCardUrl = getProductUrl(codeOrId, title);

  const imageSource =
    product?.image ||
    product?.images?.[0]?.medium ||
    product?.images?.[0]?.url ||
    product?.images?.[0] ||
    null;

  return (
    <div ref={impressionRef} className={cn('w-full min-w-0 select-none transition-all', className)}>
      
      {/* ================= نسخه دسکتاپ ================= */}
      <Link
        href={productCardUrl}
        prefetch={false}
        onClick={handleCardClick}
        className={cn(
          'relative hidden h-full w-full select-none flex-col rounded-lg border border-border bg-card p-3 transition-all duration-normal sm:p-3.5 md:flex',
          'hover:border-primary/40 hover:shadow-sm'
        )}
        draggable={false}
      >
        <div className="relative mb-2 aspect-[4/3] w-full select-none overflow-hidden rounded-md" draggable={false}>
          <Image
            src={getFullUrl(imageSource)}
            alt={product?.imageAlt || title}
            fill
            sizes="(max-width: 640px) 50vw, (max-width: 1024px) 33vw, 250px"
            className="select-none rounded-md object-contain"
            draggable={false}
          />
        </div>

        <div className="mb-1 mt-2 h-9 w-full">
          <h4 className="line-clamp-2 text-sm font-semibold leading-relaxed text-foreground">
            {title}
          </h4>
        </div>

        {showRating ? (
          <div className="mt-1 flex w-full select-none items-center justify-start gap-1">
            <Star aria-hidden="true" className="h-3.5 w-3.5 shrink-0 fill-warning text-warning" />
            <span className="text-2xs font-bold text-foreground sm:text-xs">
              {formatPrice(ratingValue)}
            </span>
            {ratingCount > 0 ? (
              <span className="text-2xs text-muted-foreground">
                ({toPersianDigits(ratingCount)})
              </span>
            ) : null}
          </div>
        ) : null}

        {/* بخش اسلایدر اختصاصی بین shopTitle, score, quantity */}
        <div className="flex w-full select-none flex-col items-stretch">
          {isMounted && tickerLength > 0 ? (
            <div className="relative mt-1.5 flex h-6 w-full shrink-0 select-none items-center justify-start overflow-hidden text-2xs text-muted-foreground sm:text-xs">
              <AnimatePresence mode="wait">
                <motion.div
                  key={`desktop-ticker-${tickerIndex}`}
                  initial={{ y: 12, opacity: 0 }}
                  animate={{ y: 0, opacity: 1 }}
                  exit={{ y: -12, opacity: 0 }}
                  transition={{ duration: 0.25, ease: 'easeInOut' }}
                  className="flex h-full w-full items-center justify-start gap-1.5 truncate"
                >
                  <CurrentTickerIcon
                    aria-hidden="true"
                    className={cn(
                      'h-3.5 w-3.5 shrink-0',
                      currentItem?.isShop ? 'text-shop' : 'text-primary'
                    )}
                  />
                  <span
                    className={cn(
                      'truncate font-medium',
                      currentItem?.isShop ? 'text-shop font-semibold' : 'text-foreground'
                    )}
                  >
                    {currentItem?.text}
                  </span>
                </motion.div>
              </AnimatePresence>
            </div>
          ) : (
            <div className="mt-1.5 h-6 w-full shrink-0" />
          )}
        </div>

        <div
          className={cn(
            'mt-auto flex w-full items-center justify-between pt-2',
            isOutOfStock || !showAddToBasket
              ? 'pb-1'
              : 'border-b border-dashed border-border pb-3.5'
          )}
        >
          <div
            className={cn(
              'shrink-0 rounded-sm border border-accent bg-accent px-2.5 py-1 text-xs font-bold text-accent-foreground transition-opacity duration-fast',
              hasDiscount && discountPercent > 0 ? 'opacity-100' : 'pointer-events-none opacity-0'
            )}
          >
            {toPersianDigits(discountPercent)}٪
          </div>

          <div className="flex min-w-0 flex-col items-end">
            {hasDiscount && originalPriceToman > finalPriceToman ? (
              <span className="text-2xs font-medium text-muted-foreground line-through sm:text-xs">
                {formatPrice(originalPriceToman)}
              </span>
            ) : null}
            <div className="mt-0.5 flex items-center gap-0.5">
              {isOutOfStock ? (
                <span className="text-sm font-bold text-destructive sm:text-base">ناموجود</span>
              ) : (
                <>
                  <span className="text-base font-bold text-foreground sm:text-lg">
                    {formatPrice(finalPriceToman)}
                  </span>
                  <span className="text-2xs text-muted-foreground">تومان</span>
                </>
              )}
            </div>
          </div>
        </div>

        {showAddToBasket && !isOutOfStock ? (
          <div className="mt-3 h-9 w-full shrink-0">
            {isInBasket ? (
              <div className="flex h-full items-center justify-between gap-1 rounded-md border border-primary bg-card p-1 shadow-xs">
                <button
                  type="button"
                  onClick={handleIncrease}
                  disabled={basketQuantity >= maxLimit || activeLoading}
                  aria-label="افزایش تعداد"
                  className="rounded-sm p-1 text-primary transition-colors hover:bg-muted disabled:opacity-30"
                >
                  <Plus aria-hidden="true" className="h-4 w-4" />
                </button>
                <span className="text-xs font-bold text-primary">
                  {activeLoading ? (
                    <Loader2 aria-hidden="true" className="mx-auto h-4 w-4 animate-spin text-primary" />
                  ) : (
                    new Intl.NumberFormat('fa-IR').format(basketQuantity)
                  )}
                </span>
                <button
                  type="button"
                  onClick={handleDecrease}
                  disabled={activeLoading}
                  aria-label="کاهش تعداد"
                  className="rounded-sm p-1 text-primary transition-colors hover:bg-muted"
                >
                  <Minus aria-hidden="true" className="h-4 w-4" />
                </button>
              </div>
            ) : (
              <Button
                variant="outline"
                size="sm"
                onClick={handleAddToBasket}
                isLoading={activeLoading}
                className="h-full w-full border-primary text-xs text-primary hover:bg-accent"
              >
                <span>افزودن به سبد خرید</span>
              </Button>
            )}
          </div>
        ) : null}
      </Link>

      {/* ================= نسخه موبایل ================= */}
      <Link
        href={productCardUrl}
        prefetch={false}
        onClick={handleCardClick}
        className="relative flex w-full min-w-0 select-none flex-col border-b border-border bg-card px-3 py-3.5 md:hidden"
        draggable={false}
      >
        <div className="flex w-full min-w-0 items-start gap-3">
          {/* عکس محصول */}
          <div className="relative h-[96px] w-[96px] shrink-0 overflow-hidden rounded-md bg-muted">
            <Image
              src={getFullUrl(imageSource)}
              alt={product?.imageAlt || title}
              fill
              sizes="96px"
              className="select-none rounded-md object-contain"
              draggable={false}
            />
          </div>

          {/* محتوا و مشخصات محصول */}
          <div className="flex min-w-0 flex-1 flex-col justify-between self-stretch">
            <div className="w-full min-w-0">
              <h4 className="line-clamp-2 text-xs font-semibold leading-relaxed text-foreground">
                {title}
              </h4>

              {showRating ? (
                <div className="mt-1 flex select-none items-center justify-start gap-1">
                  <Star aria-hidden="true" className="h-3 w-3 shrink-0 fill-warning text-warning" />
                  <span className="text-2xs font-bold text-foreground">
                    {formatPrice(ratingValue)}
                  </span>
                  {ratingCount > 0 ? (
                    <span className="text-2xs text-muted-foreground">
                      ({toPersianDigits(ratingCount)})
                    </span>
                  ) : null}
                </div>
              ) : null}

              {/* بخش اسلایدر موبایل بین shopTitle, score, quantity */}
              {isMounted && tickerLength > 0 ? (
                <div className="relative mt-1.5 flex h-5 w-full min-w-0 shrink-0 select-none items-center justify-start overflow-hidden text-2xs text-muted-foreground">
                  <AnimatePresence mode="wait">
                    <motion.div
                      key={`mobile-ticker-${tickerIndex}`}
                      initial={{ y: 10, opacity: 0 }}
                      animate={{ y: 0, opacity: 1 }}
                      exit={{ y: -10, opacity: 0 }}
                      transition={{ duration: 0.25, ease: 'easeInOut' }}
                      className="flex h-full w-full min-w-0 items-center justify-start gap-1.5 truncate"
                    >
                      <CurrentTickerIcon
                        aria-hidden="true"
                        className={cn(
                          'h-3 w-3 shrink-0',
                          currentItem?.isShop ? 'text-shop' : 'text-primary'
                        )}
                      />
                      <span
                        className={cn(
                          'truncate text-2xs font-medium',
                          currentItem?.isShop ? 'text-shop font-semibold' : 'text-foreground'
                        )}
                      >
                        {currentItem?.text}
                      </span>
                    </motion.div>
                  </AnimatePresence>
                </div>
              ) : (
                <div className="mt-1.5 h-5 w-full shrink-0" />
              )}
            </div>

            {/* ردیف قیمت و تخفیف */}
            <div className="mt-2 flex w-full min-w-0 items-end justify-between pt-1">
              <div
                className={cn(
                  'rounded-sm border border-accent bg-accent px-2 py-0.5 text-2xs font-bold text-accent-foreground shrink-0',
                  hasDiscount && discountPercent > 0
                    ? 'opacity-100'
                    : 'pointer-events-none opacity-0'
                )}
              >
                {toPersianDigits(discountPercent)}٪
              </div>

              <div className="flex min-w-0 flex-col items-end">
                {hasDiscount && originalPriceToman > finalPriceToman ? (
                  <span className="text-2xs text-muted-foreground line-through">
                    {formatPrice(originalPriceToman)}
                  </span>
                ) : null}
                <div className="flex items-center gap-0.5">
                  {isOutOfStock ? (
                    <span className="text-xs font-bold text-destructive">ناموجود</span>
                  ) : (
                    <>
                      <span className="text-sm font-bold text-foreground">
                        {formatPrice(finalPriceToman)}
                      </span>
                      <span className="text-2xs text-muted-foreground">تومان</span>
                    </>
                  )}
                </div>
              </div>
            </div>
          </div>
        </div>

        {/* دکمه افزودن به سبد در صورت فعال بودن */}
        {showAddToBasket && !isOutOfStock ? (
          <div className="mt-3 h-8 w-full shrink-0">
            {isInBasket ? (
              <div className="flex h-full items-center justify-between gap-1 rounded-md border border-primary bg-card p-1 shadow-xs">
                <button
                  type="button"
                  onClick={handleIncrease}
                  disabled={basketQuantity >= maxLimit || activeLoading}
                  aria-label="افزایش تعداد"
                  className="rounded-sm p-1 text-primary transition-colors hover:bg-muted"
                >
                  <Plus aria-hidden="true" className="h-3.5 w-3.5" />
                </button>
                <span className="text-xs font-bold text-primary">
                  {activeLoading ? (
                    <Loader2 aria-hidden="true" className="mx-auto h-3.5 w-3.5 animate-spin text-primary" />
                  ) : (
                    new Intl.NumberFormat('fa-IR').format(basketQuantity)
                  )}
                </span>
                <button
                  type="button"
                  onClick={handleDecrease}
                  disabled={activeLoading}
                  aria-label="کاهش تعداد"
                  className="rounded-sm p-1 text-primary transition-colors hover:bg-muted"
                >
                  <Minus aria-hidden="true" className="h-3.5 w-3.5" />
                </button>
              </div>
            ) : (
              <Button
                variant="outline"
                size="sm"
                onClick={handleAddToBasket}
                isLoading={activeLoading}
                className="h-full w-full border-primary text-xs text-primary hover:bg-accent"
              >
                <span>افزودن به سبد خرید</span>
              </Button>
            )}
          </div>
        ) : null}
      </Link>
    </div>
  );
}

'use client';

import * as React from 'react';
import Link from 'next/link';
import { ChevronLeft, ChevronRight, Wrench } from 'lucide-react';
import { motion, useMotionValue, animate } from 'framer-motion';
import { ProductStandardCard } from './ProductStandardCard';
import { Typography } from '@/components/primitives/Typography';
import { SliderSkeleton } from './SliderSkeleton';
import { cn } from '@/design-system/utils/cn';
import { toPersianDigits } from '@/core/utils/formatters';

export interface ProductSliderProps {
  title: string;
  products: any[];
  isLoading: boolean;
  isError: boolean;
  viewAllLink?: string;
  className?: string;
  showTimer?: boolean;
  totalSalesCount?: number; // ✅ تعداد کل فروشندگان / محصولات
}

export function ProductSlider({
  title,
  products = [],
  isLoading,
  isError,
  viewAllLink = '#',
  className,
  totalSalesCount,
}: ProductSliderProps) {
  const carouselRef = React.useRef<HTMLDivElement>(null);
  const [dragWidth, setDragWidth] = React.useState(0);
  const x = useMotionValue(0);
  const [isDragging, setIsDragging] = React.useState(false);

  React.useEffect(() => {
    if (carouselRef.current) {
      setDragWidth(carouselRef.current.scrollWidth - carouselRef.current.offsetWidth);
    }
  }, [products]);

  const handleScroll = (direction: 'left' | 'right') => {
    const step = 450;
    let newX = x.get() + (direction === 'left' ? step : -step);
    if (newX < 0) newX = 0;
    if (newX > dragWidth) newX = dragWidth;
    animate(x, newX, { type: 'spring', stiffness: 200, damping: 30 });
  };

  const handleDragStart = () => {
    setIsDragging(true);
  };

  const handleDragEnd = () => {
    setTimeout(() => {
      setIsDragging(false);
    }, 50);
  };

  const handleDragClickCapture = (e: React.MouseEvent) => {
    if (isDragging) {
      e.preventDefault();
      e.stopPropagation();
    }
  };

  if (isLoading) {
    return <SliderSkeleton title={title} />;
  }

  if (isError || products.length === 0) return null;

  return (
    <div className={cn('flex w-full flex-col space-y-4 py-4', className)}>
      {/* هدر اسلایدر */}
      <div className="flex w-full items-center justify-between px-1">
        <div className="flex items-center gap-2.5">
          <div className="flex h-8 w-8 items-center justify-center rounded-lg bg-primary/10 text-primary">
            <Wrench aria-hidden="true" className="h-4 w-4 shrink-0" />
          </div>
          <div className="flex items-center gap-2">
            <Typography variant="h4" className="font-bold text-foreground">
              {title}
            </Typography>
            {totalSalesCount !== undefined && totalSalesCount > 0 && (
              <span className="rounded-md bg-muted px-2 py-0.5 text-2xs font-bold text-muted-foreground tabular-nums">
                {toPersianDigits(totalSalesCount)} کالا
              </span>
            )}
          </div>
        </div>

        {/* دکمه مشاهده همه با رنگ آبی درخواستی */}
        <Link
          href={viewAllLink}
          prefetch={false}
          className="group inline-flex shrink-0 items-center gap-1 text-xs font-semibold text-blue-600 transition-colors hover:text-blue-700 dark:text-blue-400 sm:text-sm"
        >
          <span>مشاهده همه</span>
          <ChevronLeft aria-hidden="true" className="h-4 w-4 stroke-[2.25] transition-transform duration-fast group-hover:-translate-x-1" />
        </Link>
      </div>

      {/* کانتینر اسلایدر */}
      <div className="group relative w-full overflow-hidden rounded-xl border border-border/70 bg-card p-4 shadow-xs">
        {products.length > 4 ? (
          <>
            <button
              type="button"
              onClick={() => handleScroll('right')}
              className="absolute start-3 top-1/2 z-20 hidden -translate-y-1/2 cursor-pointer items-center justify-center rounded-full border border-border/80 bg-card p-2.5 text-foreground shadow-md transition-all duration-fast hover:border-primary/50 hover:bg-primary hover:text-primary-foreground focus-visible:outline-none focus-visible:ring-2 focus-visible:ring-ring md:flex"
              aria-label="محصولات قبلی"
            >
              <ChevronRight aria-hidden="true" className="h-4 w-4" />
            </button>

            <button
              type="button"
              onClick={() => handleScroll('left')}
              className="absolute end-3 top-1/2 z-20 hidden -translate-y-1/2 cursor-pointer items-center justify-center rounded-full border border-border/80 bg-card p-2.5 text-foreground shadow-md transition-all duration-fast hover:border-primary/50 hover:bg-primary hover:text-primary-foreground focus-visible:outline-none focus-visible:ring-2 focus-visible:ring-ring md:flex"
              aria-label="محصولات بعدی"
            >
              <ChevronLeft aria-hidden="true" className="h-4 w-4" />
            </button>
          </>
        ) : null}

        <div ref={carouselRef} className="relative z-10 w-full select-none overflow-hidden">
          <motion.div
            drag="x"
            style={{ x, touchAction: 'pan-y' }}
            dragConstraints={{ left: 0, right: dragWidth }}
            dragElastic={0.12}
            dragMomentum
            onDragStart={handleDragStart}
            onDragEnd={handleDragEnd}
            dragTransition={{ power: 0.2, bounceStiffness: 200, bounceDamping: 25 }}
            onClickCapture={handleDragClickCapture}
            className="flex cursor-grab gap-4 py-1 active:cursor-grabbing"
          >
            {products.map((prod: any) => (
              <div key={prod.id} className="w-[190px] shrink-0 sm:w-[250px]">
                <ProductStandardCard product={prod} />
              </div>
            ))}
          </motion.div>
        </div>
      </div>
    </div>
  );

export interface ProductStandardCardProps {

  className?: string;

export function ProductStandardCard({
  product,

  const [tickerIndex, setTickerIndex] = React.useState(0);
  const [isMounted, setIsMounted] = React.useState(false);
  const [isDragging, setIsDragging] = React.useState(false);
  const dragStart = React.useRef({ x: 0, y: 0 });

  React.useEffect(() => {
    setIsMounted(true);
  }, []);

  const nominated = product?.nominatedShopProduct || {};

  const shopProductId = product?.shopProductId || nominated?.id || product?.id || null;
  const impressionRef = useImpression(shopProductId);

  const title =
    product?.productTitle ||
    product?.title ||
    product?.name?.value ||
    product?.name ||
    '';

  const shopName = product?.shopTitle || nominated?.shopTitle || product?.shop?.name || '';
  const scoreValue = product?.score ?? nominated?.score ?? product?.averageRate ?? 5;
  const quantityValue = product?.quantity ?? nominated?.quantity ?? 0;

  const originalPriceRaw = Number(
    product?.rialRetailPrice ??
      nominated?.rialRetailPrice ??
      product?.price?.raw ??
      (typeof product?.price === 'number' ? product.price : 0) ??
      nominated?.price ??
      0
  );

  const finalPriceRaw = Number(
    product?.rialFinalPrice ??
      nominated?.rialFinalPrice ??
      product?.discount?.discountedPriceRaw ??
      product?.discountPrice ??
      nominated?.discountPrice ??
      product?.price?.raw ??
      (typeof product?.price === 'number' ? product.price : 0) ??
      originalPriceRaw
  );

  const originalPriceToman = Math.round(originalPriceRaw / 10);
  const finalPriceToman = Math.round(finalPriceRaw / 10);

  const isOutOfStock =
    finalPriceRaw === 0 || (product?.quantity !== undefined && product.quantity === 0);

  const hasDiscount =
    product?.isDiscountApplied !== undefined
      ? product.isDiscountApplied
      : originalPriceRaw > finalPriceRaw && !isOutOfStock;

  const discountPercent =
    product?.discountPercentage ??
    (hasDiscount && originalPriceRaw > 0
      ? Math.round(((originalPriceRaw - finalPriceRaw) / originalPriceRaw) * 100)
      : nominated?.discountPercentage || product?.discount?.percent || 0);

  const tickerItems = React.useMemo(() => {
    const items: { text: string; icon: any; isShop?: boolean }[] = [];

    items.push({
      text: shopName ? `فروشگاه: ${shopName}` : 'فروشگاه: یدکچی',
      icon: Store,
      isShop: true,
    });

    items.push({
      text: `امتیاز: ${toPersianDigits(scoreValue)} از ۵`,

    });

      text: quantityValue > 0 ? `موجودی در انبار: ${toPersianDigits(quantityValue)} عدد` : 'وضعیت: ناموجود در انبار',
      icon: Package,
    });

    return items;
  }, [shopName, scoreValue, quantityValue]);

  const tickerLength = tickerItems.length;

  React.useEffect(() => {
    if (!isMounted || tickerLength <= 1) return;
    const interval = setInterval(() => {
      setTickerIndex((prev) => (prev + 1) % tickerLength);
    }, 3000);
    return () => clearInterval(interval);
  }, [tickerLength, isMounted]);

  const formatPrice = (value: number) => new Intl.NumberFormat('fa-IR').format(value);

  const handleMouseDown = (e: React.MouseEvent) => {
    dragStart.current = { x: e.clientX, y: e.clientY };
    setIsDragging(false);
  };

  const handleMouseMove = (e: React.MouseEvent) => {
    const dx = Math.abs(e.clientX - dragStart.current.x);
    const dy = Math.abs(e.clientY - dragStart.current.y);
    if (dx > 6 || dy > 6) setIsDragging(true);
  };

  const handleTouchStart = (e: React.TouchEvent) => {
    const touch = e.touches[0];
    if (touch) {
      dragStart.current = { x: touch.clientX, y: touch.clientY };
      setIsDragging(false);
    }
  };

  const handleTouchMove = (e: React.TouchEvent) => {
    const touch = e.touches[0];
    if (touch) {
      const dx = Math.abs(touch.clientX - dragStart.current.x);
      const dy = Math.abs(touch.clientY - dragStart.current.y);
      if (dx > 6 || dy > 6) setIsDragging(true);
    }
  };

  const handleClick = (e: React.MouseEvent) => {
    if (isDragging) {
      e.preventDefault();
      e.stopPropagation();

          {formatPrice(rating)}
        </span>
      </div>
    );
  };

  const currentItem = tickerItems[tickerIndex];

    product?.productCode || product?.code || product?.productId || product?.id;
  const productCardUrl = getProductUrl(codeOrId, title);

    product?.images?.[0]?.medium ||
    product?.images?.[0]?.url ||
    product?.images?.[0] ||
    null;

  return (
    <Link
      href={productCardUrl}
      prefetch={false}
      className="block h-full w-full select-none"
      draggable={false}

      onTouchStart={handleTouchStart}

      onClick={handleClick}
    >
      <div
        ref={impressionRef}
        className={cn(
          'relative flex h-full w-full select-none flex-col items-center rounded-lg border border-border bg-card p-3 transition-all duration-normal sm:p-3.5',
          'hover:border-primary/50 hover:shadow-md',
          className
        )}
      >
        <div className="relative mb-2 aspect-[4/3] w-full select-none overflow-hidden rounded-md" draggable={false}>
          <Image
            src={getFullUrl(imageSource)}
            alt={product?.imageAlt || title}
            fill
            sizes="(max-width: 640px) 50vw, (max-width: 1024px) 33vw, 250px"
            className="select-none rounded-md object-contain"
            draggable={false}
          />

          {showRating ? (
            <div className="absolute start-2 top-2 z-10 flex items-center justify-center rounded-md bg-card/90 px-2 py-0.5 shadow-sm backdrop-blur-sm">

              <Filter className="h-4.5 w-4.5 text-primary" />
              <Typography variant="h5" className="font-extrabold text-foreground">
                فیلترها
              </Typography>
            </div>
            {hasActiveFilters && (
              <button
                type="button"
                onClick={onClearAll}
                className="text-xs font-bold text-destructive transition-colors hover:underline"
              >
                حذف فیلترها
              </button>
            )}
          </div>
          {renderFilterContent(false)}
        </div>
      </div>

      {/* مودال فیلترهای موبایل */}
      <Modal
        isOpen={isOpen}
        onClose={onClose}
        className="fixed inset-0 z-modal flex h-full max-h-full w-full max-w-none flex-col rounded-none bg-card p-0"
        overlayClassName="bg-black/40 backdrop-blur-sm"
      >
        <div className="flex shrink-0 items-center justify-between border-b border-border bg-muted/20 px-4 py-3.5" dir="rtl">
          <div className="flex items-center gap-2">
            <button
              type="button"
              onClick={onClose}
              className="rounded-full p-1 text-foreground transition-colors hover:bg-muted"
              aria-label="بستن"
            >
              <ArrowRight className="h-5 w-5" />
            </button>
            <span className="flex items-center gap-1.5 text-sm font-bold text-foreground">
              <Filter className="h-4 w-4 text-primary" />
              فیلترها
            </span>
          </div>
          <button
            type="button"
            onClick={onClose}
            className="flex items-center justify-center rounded-full p-1.5 text-muted-foreground transition-colors hover:bg-muted hover:text-foreground"
            aria-label="بستن"
          >
            <X className="h-4 w-4" />
          </button>
        </div>

        <div className="relative flex-1 overflow-y-auto p-5" dir="rtl">
          {renderFilterContent(true)}
        </div>

        <div className="absolute bottom-0 right-0 left-0 z-sticky flex shrink-0 gap-3 border-t border-border bg-card p-4 shadow-md" dir="rtl">
          {hasActiveFilters && (
            <Button
              variant="outline"
              size="sm"
              onClick={() => {
                onClearAll();
                onClose();
              }}
              className="h-10 flex-1 rounded-md border-destructive/20 text-xs font-bold text-destructive hover:bg-destructive/5"
            >
              حذف فیلترها
            </Button>
          )}
          <Button
            variant="primary"
            size="sm"
            onClick={onClose}
            className="h-10 flex-1 rounded-md text-xs font-bold"
          >
            مشاهده نتایج
          </Button>
        </div>
      </Modal>
    </>
  );
}
