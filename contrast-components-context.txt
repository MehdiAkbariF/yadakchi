export interface ModalProps {
  isOpen: boolean;
  onClose: () => void;
  children: React.ReactNode;
  className?: string;
  overlayClassName?: string;
  closeOnOverlayClick?: boolean;
  closeOnEscape?: boolean;
}

export function Modal({
  isOpen,
  onClose,
  children,
  className,
  overlayClassName,
  closeOnOverlayClick = true,
  closeOnEscape = true,
}: ModalProps) {
  const modalRef = React.useRef<HTMLDivElement>(null);
  const [mounted, setMounted] = React.useState(false);

  React.useEffect(() => {
    setMounted(true);
  }, []);

  React.useEffect(() => {
    if (isOpen) {
      document.body.style.overflow = 'hidden';
      modalRef.current?.focus();
    }
    return () => {
      document.body.style.overflow = '';
    };
  }, [isOpen]);

  React.useEffect(() => {
    if (!isOpen || !closeOnEscape) return;
    const handleEscape = (e: KeyboardEvent) => {
      if (e.key === 'Escape') onClose();
    };
    document.addEventListener('keydown', handleEscape);
    return () => document.removeEventListener('keydown', handleEscape);
  }, [isOpen, closeOnEscape, onClose]);

  if (!isOpen || !mounted) return null;

  return createPortal(
    <div
      className="fixed inset-0 z-modal flex items-center justify-center"
      role="dialog"
      aria-modal="true"
    >
      <div
        className={cn(
          'fixed inset-0 bg-foreground/40 backdrop-blur-sm animate-fade-in',
          overlayClassName
        )}
        onClick={closeOnOverlayClick ? onClose : undefined}
        aria-hidden="true"
      />
      <div
        ref={modalRef}
        className={cn(
          'relative z-10 flex flex-col bg-background overflow-hidden',
          'w-full h-full max-h-full max-w-none rounded-none',
          'md:w-full md:max-w-2xl md:h-auto md:max-h-[90vh] md:rounded-xl md:shadow-lg',
          className
        )}
        tabIndex={-1}
      >
        {children}
      </div>
    </div>,
    document.body
  );
}

import * as React from 'react';

import { cn } from '@/design-system/utils/cn';

export type ToastVariant = 'success' | 'error' | 'warning' | 'info';

export interface ToastProps {
  message: string;
  variant?: ToastVariant;
  duration?: number;
  onClose?: () => void;
  className?: string;
}

const iconMap: Record<ToastVariant, React.ReactNode> = {
  success: <CheckCircle aria-hidden="true" className="h-5 w-5 text-success" />,
  error: <AlertCircle aria-hidden="true" className="h-5 w-5 text-destructive" />,
  warning: <AlertTriangle aria-hidden="true" className="h-5 w-5 text-warning" />,
  info: <Info aria-hidden="true" className="h-5 w-5 text-info" />,
};

const variantStyles: Record<ToastVariant, string> = {
  success: 'border-success bg-card',
  error: 'border-destructive bg-card',
  warning: 'border-warning bg-card',
  info: 'border-info bg-card',
};

export function Toast({ message, variant = 'info', duration = 5000, onClose, className }: ToastProps) {
  const [isVisible, setIsVisible] = React.useState(true);

  React.useEffect(() => {
    const timer = setTimeout(() => {
      setIsVisible(false);
      onClose?.();
    }, duration);
    return () => clearTimeout(timer);
  }, [duration, onClose]);

  if (!isVisible) return null;

  return (
    <div
      role="alert"
      className={cn(
        'flex items-center gap-3 rounded-lg border p-4 shadow-md',
        'animate-slide-in-from-top',
        variantStyles[variant],
        className
      )}
    >
      {iconMap[variant]}
      <span className="flex-1 text-sm text-foreground">{message}</span>
      <Button
        variant="ghost"
        size="icon-sm"
        onClick={() => {
          setIsVisible(false);
          onClose?.();
        }}
        className="shrink-0"
      >
        <X aria-hidden="true" className="h-4 w-4" />
        <span className="sr-only">بستن</span>
      </Button>
    </div>
  );
}

'use client';

import * as React from 'react';
import Link from 'next/link';
import Image from 'next/image';
import { Star, Store, Package } from 'lucide-react';
import { motion, AnimatePresence } from 'framer-motion';
import { cn } from '@/design-system/utils/cn';
import { getProductUrl, toPersianDigits, getFullUrl } from '@/core/utils/formatters';
import { useImpression } from '@/shared/hooks/useImpression';
import { trackShopProductClick } from '@/core/utils/impression-tracker';

export interface ProductStandardCardProps {
  product: any;
  showRating?: boolean;
  className?: string;
}

export function ProductStandardCard({
  product,
  showRating = true,
  className,
}: ProductStandardCardProps) {
  const [tickerIndex, setTickerIndex] = React.useState(0);
  const [isMounted, setIsMounted] = React.useState(false);
  const [isDragging, setIsDragging] = React.useState(false);
  const dragStart = React.useRef({ x: 0, y: 0 });

  React.useEffect(() => {
    setIsMounted(true);
  }, []);

  const nominated = product?.nominatedShopProduct || {};

  const shopProductId = product?.shopProductId || nominated?.id || product?.id || null;
  const impressionRef = useImpression(shopProductId);

  const title =
    product?.productTitle ||
    product?.title ||
    product?.name?.value ||
    product?.name ||
    '';

  // استخراج ۳ مقدار مورد نظر برای اسلایدر
  const shopName = product?.shopTitle || nominated?.shopTitle || product?.shop?.name || '';
  const scoreValue = product?.score ?? nominated?.score ?? product?.averageRate ?? 5;
  const quantityValue = product?.quantity ?? nominated?.quantity ?? 0;

  const originalPriceRaw = Number(
    product?.rialRetailPrice ??
      nominated?.rialRetailPrice ??
      product?.price?.raw ??
      (typeof product?.price === 'number' ? product.price : 0) ??
      nominated?.price ??
      0
  );

  const finalPriceRaw = Number(
    product?.rialFinalPrice ??
      nominated?.rialFinalPrice ??
      product?.discount?.discountedPriceRaw ??
      product?.discountPrice ??
      nominated?.discountPrice ??
      product?.price?.raw ??
      (typeof product?.price === 'number' ? product.price : 0) ??
      originalPriceRaw
  );

  const originalPriceToman = Math.round(originalPriceRaw / 10);
  const finalPriceToman = Math.round(finalPriceRaw / 10);

  const isOutOfStock =
    finalPriceRaw === 0 || (product?.quantity !== undefined && product.quantity === 0);

  const hasDiscount =
    product?.isDiscountApplied !== undefined
      ? product.isDiscountApplied
      : originalPriceRaw > finalPriceRaw && !isOutOfStock;

  const discountPercent =
    product?.discountPercentage ??
    (hasDiscount && originalPriceRaw > 0
      ? Math.round(((originalPriceRaw - finalPriceRaw) / originalPriceRaw) * 100)
      : nominated?.discountPercentage || product?.discount?.percent || 0);

  // تنظیم دقیق اسلایدر فقط بین: نام فروشگاه، امتیاز و موجودی
  const tickerItems = React.useMemo(() => {
    const items: { text: string; icon: any; isShop?: boolean }[] = [];

    // ۱. نام فروشگاه
    items.push({
      text: shopName ? `فروشگاه: ${shopName}` : 'فروشگاه: یدکچی',
      icon: Store,
      isShop: true,
    });

    // ۲. امتیاز (Score)
    items.push({
      text: `امتیاز: ${toPersianDigits(scoreValue)} از ۵`,
      icon: Star,
    });

    // ۳. موجودی (Quantity)
    items.push({
      text: quantityValue > 0 ? `موجودی در انبار: ${toPersianDigits(quantityValue)} عدد` : 'وضعیت: ناموجود در انبار',
      icon: Package,
    });

    return items;
  }, [shopName, scoreValue, quantityValue]);

  const tickerLength = tickerItems.length;

  React.useEffect(() => {
    if (!isMounted || tickerLength <= 1) return;
    const interval = setInterval(() => {
      setTickerIndex((prev) => (prev + 1) % tickerLength);
    }, 3000);
    return () => clearInterval(interval);
  }, [tickerLength, isMounted]);

  const formatPrice = (value: number) => new Intl.NumberFormat('fa-IR').format(value);

  const handleMouseDown = (e: React.MouseEvent) => {
    dragStart.current = { x: e.clientX, y: e.clientY };
    setIsDragging(false);
  };

  const handleMouseMove = (e: React.MouseEvent) => {
    const dx = Math.abs(e.clientX - dragStart.current.x);
    const dy = Math.abs(e.clientY - dragStart.current.y);
    if (dx > 6 || dy > 6) setIsDragging(true);
  };

  const handleTouchStart = (e: React.TouchEvent) => {
    const touch = e.touches[0];
    if (touch) {
      dragStart.current = { x: touch.clientX, y: touch.clientY };
      setIsDragging(false);
    }
  };

  const handleTouchMove = (e: React.TouchEvent) => {
    const touch = e.touches[0];
    if (touch) {
      const dx = Math.abs(touch.clientX - dragStart.current.x);
      const dy = Math.abs(touch.clientY - dragStart.current.y);
      if (dx > 6 || dy > 6) setIsDragging(true);
    }
  };

  const handleClick = (e: React.MouseEvent) => {
    if (isDragging) {
      e.preventDefault();
      e.stopPropagation();
      return;
    }
    if (shopProductId) trackShopProductClick(shopProductId);
  };

  const renderStars = () => {
    const rating = product?.averageRate || product?.rating?.average || scoreValue;
    return (
      <div className="flex select-none items-center gap-1">
        <Star aria-hidden="true" className="h-3 w-3 shrink-0 fill-warning text-warning" />
        <span className="text-2xs font-bold text-foreground sm:text-xs">
          {formatPrice(rating)}
        </span>
      </div>
    );
  };

  const CurrentTickerIcon = tickerItems[tickerIndex]?.icon || Store;
  const currentItem = tickerItems[tickerIndex];

  const codeOrId =
    product?.productCode || product?.code || product?.productId || product?.id;
  const productCardUrl = getProductUrl(codeOrId, title);

  const imageSource =
    product?.image ||
    product?.images?.[0]?.medium ||
    product?.images?.[0]?.url ||
    product?.images?.[0] ||
    null;

  return (
    <Link
      href={productCardUrl}
      prefetch={false}
      className="block h-full w-full select-none"
      draggable={false}
      onMouseDown={handleMouseDown}
      onMouseMove={handleMouseMove}
      onTouchStart={handleTouchStart}
      onTouchMove={handleTouchMove}
      onClick={handleClick}
    >
      <div
        ref={impressionRef}
        className={cn(
          'relative flex h-full w-full select-none flex-col items-center rounded-lg border border-border bg-card p-3 transition-all duration-normal sm:p-3.5',
          'hover:border-primary/40 hover:shadow-sm',
          className
        )}
      >
        <div className="relative mb-2 aspect-[4/3] w-full select-none overflow-hidden rounded-md" draggable={false}>
          <Image
            src={getFullUrl(imageSource)}
            alt={product?.imageAlt || title}
            fill
            sizes="(max-width: 640px) 50vw, (max-width: 1024px) 33vw, 250px"
            className="select-none rounded-md object-contain"
            draggable={false}
          />

          {showRating ? (
            <div className="absolute start-2 top-2 z-10 flex items-center justify-center rounded-sm bg-card/85 px-2 py-0.5 backdrop-blur-sm">
              {renderStars()}
            </div>
          ) : null}
        </div>

        <div className="mb-1 mt-2 h-9 w-full">
          <h4 className="line-clamp-2 w-full text-sm font-semibold leading-relaxed text-foreground">
            {title}
          </h4>
        </div>

        {/* بخش اسلایدر اختصاصی بین نام فروشگاه، امتیاز و موجودی */}
        <div className="flex w-full select-none flex-col items-stretch">
          {isMounted && tickerLength > 0 ? (
            <div className="relative mt-0.5 flex h-6 w-full shrink-0 select-none items-center justify-start overflow-hidden text-2xs text-muted-foreground sm:text-xs">
              <div className="relative hidden h-full w-full md:block">
                <AnimatePresence mode="wait">
                  <motion.div
                    key={`standard-desktop-ticker-${tickerIndex}`}
                    initial={{ y: 15, opacity: 0 }}
                    animate={{ y: 0, opacity: 1 }}
                    exit={{ y: -15, opacity: 0 }}
                    transition={{ type: 'spring', stiffness: 300, damping: 22 }}
                    className="absolute inset-x-0 bottom-0 top-0 flex h-full items-center justify-start gap-1.5 truncate py-1"
                  >
                    <CurrentTickerIcon
                      aria-hidden="true"
                      className={cn(
                        'h-3.5 w-3.5 shrink-0',
                        currentItem?.isShop ? 'text-shop' : 'text-primary'
                      )}
                    />
                    <span
                      className={cn(
                        'truncate font-medium',
                        currentItem?.isShop ? 'text-shop font-semibold' : 'text-foreground'
                      )}
                    >
                      {currentItem?.text}
                    </span>
                  </motion.div>
                </AnimatePresence>
              </div>

              {/* نسخه موبایل (اسلایدر برای موبایل نیز فعال شد) */}
              <div className="relative flex h-full w-full items-center justify-start gap-1.5 truncate md:hidden">
                <AnimatePresence mode="wait">
                  <motion.div
                    key={`standard-mobile-ticker-${tickerIndex}`}
                    initial={{ y: 10, opacity: 0 }}
                    animate={{ y: 0, opacity: 1 }}
                    exit={{ y: -10, opacity: 0 }}
                    transition={{ duration: 0.25, ease: 'easeInOut' }}
                    className="absolute inset-x-0 bottom-0 top-0 flex h-full items-center justify-start gap-1.5 truncate"
                  >
                    <CurrentTickerIcon
                      aria-hidden="true"
                      className={cn(
                        'h-3 w-3 shrink-0',
                        currentItem?.isShop ? 'text-shop' : 'text-primary'
                      )}
                    />
                    <span
                      className={cn(
                        'truncate text-2xs font-medium',
                        currentItem?.isShop ? 'text-shop font-semibold' : 'text-foreground'
                      )}
                    >
                      {currentItem?.text}
                    </span>
                  </motion.div>
                </AnimatePresence>
              </div>
            </div>
          ) : (
            <div className="mt-0.5 h-6 w-full shrink-0" />
          )}
        </div>

        <div className="mt-auto flex w-full items-center justify-between pt-2">
          <div
            className={cn(
              'shrink-0 rounded-sm border border-accent bg-accent px-2.5 py-1 text-xs font-bold text-accent-foreground transition-opacity duration-fast',
              hasDiscount && discountPercent > 0 ? 'opacity-100' : 'pointer-events-none opacity-0'
            )}
          >
            {toPersianDigits(discountPercent)}٪
          </div>

          <div className="flex min-w-0 flex-col items-end">
            {hasDiscount && originalPriceToman > finalPriceToman ? (
              <span className="text-2xs font-medium text-muted-foreground line-through sm:text-xs">
                {formatPrice(originalPriceToman)}
              </span>
            ) : null}
            <div className="mt-0.5 flex items-center gap-0.5">
              {isOutOfStock ? (
                <span className="text-sm font-bold text-destructive sm:text-base">ناموجود</span>
              ) : (
                <>
                  <span className="text-base font-bold text-foreground sm:text-lg">
