export interface HeaderProps {
  className?: string;

// مسیرهای flow: در موبایل هدر اختصاصی می‌گیرند
const FLOW_PAGES_MAP: Record<string, FlowStep> = {
  '/basket': 'basket',
  '/checkout': 'checkout',
  '/basket-payment': 'payment',
};

function getFlowStep(pathname: string | null): FlowStep | null {
  if (!pathname) return null;
  for (const [path, step] of Object.entries(FLOW_PAGES_MAP)) {
    if (pathname === path || pathname.startsWith(`${path}/`)) return step;
  }
  return null;
}

export function Header({ className }: HeaderProps) {
  const router = useRouter();
  const pathname = usePathname();
  useAuth();
  const isHeaderMinimized = useAppStore((state) => state.isHeaderMinimized);
  const setIsHeaderMinimized = useAppStore((state) => state.setIsHeaderMinimized);
  const { scrollY } = useScroll();

  const flowStep = getFlowStep(pathname);
  const isFlowPage = !!flowStep;

  React.useEffect(() => {
    setIsHeaderMinimized(false);
  }, [pathname, setIsHeaderMinimized]);

  useMotionValueEvent(scrollY, 'change', (latest) => {
    const prev = scrollY.getPrevious() ?? 0;
    const diff = latest - prev;

    if (latest < 40) {
      if (isHeaderMinimized) setIsHeaderMinimized(false);
      return;
    }

    if (diff < -8) {
      if (isHeaderMinimized) setIsHeaderMinimized(false);
    } else if (diff > 10 && latest > 80) {
      if (!isHeaderMinimized) setIsHeaderMinimized(true);
    }
  });

  const handleSearch = (query: string) => {
    if (query.trim()) {
      router.push(`/search?q=${encodeURIComponent(query)}`);
    }
  };

  React.useEffect(() => {
    if (typeof window !== 'undefined' && window.innerWidth >= 1024) {
      document.documentElement.style.setProperty(
        '--header-height',
        isHeaderMinimized ? '4rem' : '6.75rem'
      );
    }
  }, [isHeaderMinimized]);

  return (
    <header className={cn('z-40 w-full select-none', className)}>
      {/* ============ هدر دسکتاپ ============ */}
      <div className="pointer-events-none relative hidden h-[6.75rem] w-full lg:block">
        <motion.div
          className="pointer-events-auto fixed inset-x-0 top-0 z-40 flex w-full flex-col border-b border-border/60 bg-card/95 shadow-xs backdrop-blur-md"
          initial={false}
          animate={isHeaderMinimized ? 'collapsed' : 'expanded'}
          variants={{
            expanded: { height: '6.75rem', transition: { duration: 0.2, ease: 'easeInOut' } },
            collapsed: { height: '4rem', transition: { duration: 0.2, ease: 'easeInOut' } },
          }}
        >
          <div className="flex h-16 w-full items-center px-4 lg:px-8">
            <div className="mx-auto flex w-full max-w-[1400px] items-center justify-between gap-4">
              <Logo className="shrink-0 pt-0" />

              <div className="mx-2 max-w-2xl flex-1 xl:max-w-3xl">
                <SearchBar
                  placeholder="جستجوی هوشمند قطعات..."
                  onSearch={handleSearch}
                  isMobile={false}
                />
              </div>

              <div className="flex shrink-0 items-center gap-3">
                <SellerButton />
                <AuthButton />
                <CartButton />
                <div className="ms-1 border-s border-border/70 pe-1">
                  <ThemeToggle />
                </div>
              </div>
            </div>
          </div>

          <motion.div
            initial={false}
            animate={isHeaderMinimized ? 'collapsed' : 'expanded'}
            variants={{
              expanded: { height: '2.75rem', opacity: 1, transition: { duration: 0.2, ease: 'easeInOut' } },
              collapsed: { height: 0, opacity: 0, transition: { duration: 0.15, ease: 'easeInOut' } },
            }}
            className={cn(
              'w-full border-t border-border/50 bg-secondary/30',
              isHeaderMinimized ? 'pointer-events-none overflow-hidden' : 'overflow-visible'
            )}
          >
            <div className="mx-auto max-w-[1400px] px-4 lg:px-8">
              <div className="flex h-11 items-center justify-between">
                <DesktopNavigation />
                <div className="flex items-center gap-4">
                  <MyCarButton />
                  <CitySelector />
                </div>
              </div>
            </div>
          </motion.div>
        </motion.div>
      </div>

      {/* ============ هدر موبایل ============ */}
      {isFlowPage && flowStep ? (
        <>
          <div className="h-[5.75rem] w-full lg:hidden" />
          <FlowHeader step={flowStep} />
          <FlowProgressBar currentStep={flowStep} />
        </>
      ) : (
        <>
          <div className="h-36 w-full lg:hidden" />
          <div className="fixed inset-x-0 top-0 z-40 w-full border-b border-border/60 bg-card/95 px-4 py-2 shadow-xs backdrop-blur-md lg:hidden">
            <div className="mx-auto max-w-[1400px]">
              <MobileHeader onSearch={handleSearch} isScrolled={isHeaderMinimized} />
            </div>
          </div>
        </>
      )}
    </header>
  );
}

import { useState, useEffect, useMemo } from 'react';

import { cn } from '@/design-system/utils/cn';
import { Modal } from '@/components/composites/Modal/Modal';
import { Input } from '@/components/primitives/Input/Input';
import { useGetProvinceCitiesTree } from '@/domains/front/reference/city/hooks/city.hooks';
import { useAppStore } from '@/shared/store/useAppStore';

export interface CitySelectorProps {
  variant?: 'default' | 'mobile';
}

export function CitySelector({ variant = 'default' }: CitySelectorProps) {
  const [isOpen, setIsOpen] = useState(false);
  const [mounted, setMounted] = useState(false);
  const storeCity = useAppStore((state) => state.selectedCity);
  const setStoreCity = useAppStore((state) => state.setSelectedCity);

  const { data: provincesTree = [], isLoading } = useGetProvinceCitiesTree();
  const [searchQuery, setSearchQuery] = useState('');
  const [activeProvinceId, setActiveProvinceId] = useState<string | null>(null);

  useEffect(() => {
    setMounted(true);
  }, []);

  const selectedCity = mounted ? storeCity : null;

  const handleOpenModal = () => {
    setIsOpen(true);
    setSearchQuery('');
    setActiveProvinceId(null);
  };

  const handleSelectCity = (id: string, name: string) => {
    setStoreCity({ id, name });
    setIsOpen(false);
  };

  const handleClearCity = (e: React.MouseEvent) => {
    e.stopPropagation();
    setStoreCity(null);
  };

  const filteredCities = useMemo(() => {
    if (!searchQuery.trim()) return [];

    const results: Array<{ id: string; name: string; provinceName: string }> = [];
    provincesTree.forEach((province) => {
      province.cities.forEach((city) => {
        if (city.name.includes(searchQuery.trim())) {
          results.push({
            id: city.id,
            name: city.name,
            provinceName: province.name,
          });
        }
      });
    });
    return results;
  }, [searchQuery, provincesTree]);

  const activeProvince =
    provincesTree.find((p) => p.id === (activeProvinceId || provincesTree[0]?.id)) || provincesTree[0];
  const isMobileVariant = variant === 'mobile';

  const renderModalContent = () => (
    <Modal
      isOpen={isOpen}
      onClose={() => setIsOpen(false)}
      className="fixed inset-0 z-modal flex h-full max-h-full w-full max-w-none flex-col rounded-none bg-card p-0 md:relative md:h-[34.375rem] md:max-h-[90vh] md:max-w-2xl md:overflow-hidden md:rounded-xl md:border md:border-border md:shadow-lg"

            onClick={() => setIsOpen(false)}
            className="rounded-full p-1 text-foreground transition-colors hover:bg-muted md:hidden"
            aria-label="بازگشت به صفحه قبلی"
            type="button"
          >
            <ArrowRight aria-hidden="true" className="h-5 w-5" />
          </button>
          <span className="flex items-center gap-1.5 text-xs font-bold sm:text-sm text-foreground">
            <MapPin aria-hidden="true" className="h-4 w-4 text-primary" />
            انتخاب شهر یا استان
          </span>
        </div>
        <button
          onClick={() => setIsOpen(false)}
          className="hidden rounded-full p-1.5 text-muted-foreground transition-colors hover:bg-muted hover:text-foreground md:flex"
          aria-label="بستن پنجره"
          type="button"
        >
          <X aria-hidden="true" className="h-4 w-4" />
        </button>
      </div>

      <div className="shrink-0 border-b border-border bg-card p-4" dir="rtl">
        <label htmlFor="city-search-input" className="sr-only">جستجوی نام شهر</label>
        <Input
          id="city-search-input"
          type="text"
          placeholder="جستجوی نام شهر..."
          value={searchQuery}
          onChange={(e) => setSearchQuery(e.target.value)}
          leftIcon={<Search aria-hidden="true" className="h-4 w-4 text-muted-foreground" />}
          rightIcon={
            searchQuery ? (

                className="flex items-center justify-center rounded-full p-1 transition-colors hover:bg-muted"
                type="button"
                aria-label="پاک کردن متن جستجو"

              </button>

          }
          className="w-full border-border bg-card"
          dir="rtl"
        />
      </div>

                              ? 'border-primary bg-primary/10 font-bold text-primary'
                              : 'border-border text-foreground hover:border-primary/40 hover:bg-muted/40'

                        >
                          {city.name}

    </>
  );

'use client';

import { usePathname } from 'next/navigation';

import { Footer } from '@/components/sections/Footer';
import { BottomNav } from '@/components/sections/BottomNav/BottomNav';
import { cn } from '@/design-system/utils/cn';

export interface MainLayoutProps {
  children: React.ReactNode;
  className?: string;
  hideHeader?: boolean;
  hideFooter?: boolean;
  hideBottomNav?: boolean;
  /** مخفی کردن هدر اصلی فقط در موبایل (دسکتاپ نمایش داده می‌شود) */
  hideHeaderOnMobile?: boolean;
}

const FLOW_PAGES = ['/basket', '/checkout', '/basket-payment'];
function isFlowPage(pathname: string | null): boolean {
  if (!pathname) return false;
  return FLOW_PAGES.some((p) => pathname === p || pathname.startsWith(`${p}/`));
}

export function MainLayout({
  children,
  className,
  hideHeader = false,
  hideFooter = false,
  hideBottomNav = false,
  hideHeaderOnMobile = false,
}: MainLayoutProps) {
  const pathname = usePathname();

  const isProfileSection = pathname.startsWith('/profile');
  const isProductPage = pathname.startsWith('/product/');
  const isFlow = isFlowPage(pathname);

  return (
    <div className="flex min-h-[100dvh] flex-col bg-background">
      {!hideHeader && (
        <div
          className={cn(
            isProfileSection && 'hidden lg:block',
            isProductPage && 'hidden lg:block',
            hideHeaderOnMobile && 'hidden lg:block'
          )}
        >
          <Header />
        </div>
      )}

      <main
        className={cn(
          'flex-1 w-full max-w-[1400px] mx-auto min-h-[80vh]',
          isProfileSection
            ? 'px-0 py-0 pb-16 lg:pb-8 lg:px-6 lg:py-6'
            : isProductPage
            ? 'px-0 py-0 pb-0 lg:px-6 lg:py-6'
            : isFlow
            ? 'px-4 md:px-6 py-5 pb-44 lg:pb-6'
            : 'px-4 md:px-6 py-5 pb-20 lg:pb-6',
          className
        )}
      >
        {children}
      </main>

      {!hideFooter && (
        <div
          className={cn(
            isProfileSection && 'hidden lg:block',
            isProductPage && 'hidden lg:block'
          )}
        >
          <Footer />
        </div>
      )}

      {/* BottomNav در همه صفحات موبایل (به‌جز product) */}
      {!hideBottomNav && !isProductPage && <BottomNav />}
    </div>
  );
}
