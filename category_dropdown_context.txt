'use client';

import { useState } from 'react';
import Link from 'next/link';
import Image from 'next/image';
import { useGetMegaMenu } from '@/domains/front/banner/hooks/banner.hooks';
import {
  ChevronLeft,
  Loader2,
  AlertCircle,
  Fuel,
  Zap,
  Activity,
  Settings,
  Gauge,
  Wrench,
  Sparkles,
} from 'lucide-react';
import { cn } from '@/design-system/utils/cn';
import { Typography } from '@/components/primitives/Typography';

// ============================================================
// Types
// ============================================================
interface CategoryDropdownProps {
  isOpen: boolean;
}

interface CategoryPart {
  id: string;
  name: string;
  englishTitle?: string;
}

interface Category {
  id: string;
  name: string;
  englishTitle?: string;
  icon?: string;
  parts?: CategoryPart[];
}

// ============================================================
// Helpers
// ============================================================
function getFallbackIcon(categoryName: string) {
  const name = categoryName || '';
  if (
    name.includes('سوخت') ||
    name.includes('بنزین') ||
    name.includes('کنیستر') ||
    name.includes('باک') ||
    name.includes('پمپ')
  )
    return Fuel;
  if (
    name.includes('برق') ||
    name.includes('باتری') ||
    name.includes('الکترونیک') ||
    name.includes('شمع') ||
    name.includes('کوئل') ||
    name.includes('دینام')
  )
    return Zap;
  if (name.includes('ترمز') || name.includes('لنت') || name.includes('دیسک')) return Activity;
  if (
    name.includes('موتور') ||
    name.includes('سیلندر') ||
    name.includes('پیستون') ||
    name.includes('روغن')
  )
    return Settings;
  if (name.includes('گیربکس') || name.includes('جعبه دنده') || name.includes('کلاچ')) return Gauge;
  if (name.includes('بدنه') || name.includes('چراغ') || name.includes('سپر')) return Sparkles;
  return Wrench;
}

function getMainCategoryUrl(cat: Category): string {
  if (!cat) return '#';
  const targetSlug = encodeURIComponent(cat.englishTitle || cat.id);
  return `/part-category/${targetSlug}`;
}

function getSubCategoryUrl(cat: Category, part: CategoryPart): string {
  if (!cat || !part) return '#';
  const categorySlug = cat.id; // UUID امن
  const partSlug = encodeURIComponent(part.englishTitle || '');
  return `/parts/${categorySlug}/${partSlug}`;
}

// ============================================================
// Component
// ============================================================
export function CategoryDropdown({ isOpen }: CategoryDropdownProps) {
  const { data: rawData = [], isLoading, isError } = useGetMegaMenu();
  const [activeCategoryId, setActiveCategoryId] = useState<string | null>(null);
  const [failedImages, setFailedImages] = useState<Record<string, boolean>>({});

  if (!isOpen) return null;

  // استخراج آرایه دسته‌بندی‌ها
  const categories: Category[] = Array.isArray(rawData)
    ? rawData
    : ((rawData as any)?.partCategories as Category[]) || [];

  const activeCategory =
    categories.find((c) => c.id === (activeCategoryId || categories[0]?.id)) || categories[0];

  const handleImageError = (id: string) => {
    setFailedImages((prev) => ({ ...prev, [id]: true }));
  };

  return (
    <div
      dir="rtl"
      className="absolute top-full right-0 z-dropdown flex h-[28.125rem] w-[50rem] overflow-hidden rounded-lg border border-border bg-card shadow-lg"
    >
      {/* Loading */}
      {isLoading && (
        <div className="flex flex-1 items-center justify-center gap-2 text-muted-foreground">
          <Loader2 className="h-5 w-5 animate-spin text-primary" aria-hidden="true" />
          <Typography variant="small" weight="medium">
            در حال بارگذاری دسته‌بندی‌ها...
          </Typography>
        </div>
      )}

      {/* Error */}
      {isError && (
        <div className="flex flex-1 items-center justify-center gap-2 p-4 text-destructive">
          <AlertCircle className="h-5 w-5" aria-hidden="true" />
          <Typography variant="small" weight="medium">
            خطا در بارگذاری اطلاعات. لطفاً دوباره تلاش کنید.
          </Typography>
        </div>
      )}

      {/* Content */}
      {!isLoading && !isError && categories.length > 0 && (
        <>
          {/* ستون راست: دسته‌بندی‌های اصلی */}
          <div className="w-[15rem] shrink-0 overflow-y-auto border-l border-border bg-muted/20 py-2">
            {categories.map((category) => {
              const isActive = activeCategory?.id === category.id;
              const hasImageFailed = failedImages[category.id] || !category.icon;
              const FallbackIconComponent = getFallbackIcon(category.name);

              return (
                <button
                  key={category.id}
                  type="button"
                  onClick={() => setActiveCategoryId(category.id)}
                  onMouseEnter={() => setActiveCategoryId(category.id)}
                  aria-pressed={isActive}
                  className={cn(
                    'flex w-full items-center justify-between px-4 py-3 text-sm font-medium transition-colors text-right',
                    isActive
                      ? 'bg-card text-primary font-bold border-r-2 border-primary'
                      : 'text-foreground hover:bg-muted/50'
                  )}
                >
                  <div className="flex min-w-0 items-center gap-2.5">
                    {hasImageFailed ? (
                      <FallbackIconComponent
                        className="h-5 w-5 shrink-0 text-muted-foreground/80"
                        aria-hidden="true"
                      />
                    ) : (
                      <span className="relative inline-flex h-5 w-5 shrink-0">
                        <Image
                          src={category.icon!}
                          alt=""
                          fill
                          sizes="20px"
                          className="object-contain filter dark:invert"
                          onError={() => handleImageError(category.id)}
                        />
                      </span>
                    )}
                    <span className="truncate">{category.name}</span>
                  </div>
                  <ChevronLeft
                    aria-hidden="true"
                    className={cn(
                      'h-4 w-4 shrink-0 transition-transform',
                      isActive ? 'translate-x-1 text-primary' : 'text-muted-foreground'
                    )}
                  />
                </button>
              );
            })}
          </div>

          {/* ستون چپ: زیردسته‌ها */}
          <div className="flex-1 overflow-y-auto p-6 bg-card">
            {activeCategory ? (
              <div className="space-y-4">
                {/* هدر ستون چپ */}
                <div className="flex items-center justify-between border-b border-border pb-2">
                  <span className="flex items-center gap-2 text-title-sm">
                    {failedImages[activeCategory.id] || !activeCategory.icon ? (
                      (() => {
                        const FallbackIcon = getFallbackIcon(activeCategory.name);
                        return <FallbackIcon className="h-5 w-5 text-primary shrink-0" aria-hidden="true" />;
                      })()
                    ) : (
                      <span className="relative inline-flex h-5 w-5 shrink-0">
                        <Image
                          src={activeCategory.icon}
                          alt=""
                          fill
                          sizes="20px"
                          className="object-contain filter dark:invert"
                          onError={() => handleImageError(activeCategory.id)}
                        />
                      </span>
                    )}
                    همه قطعات {activeCategory.name}
                  </span>

                  <Link
                    href={getMainCategoryUrl(activeCategory)}
                    className="flex items-center gap-1 text-xs text-primary transition-colors hover:underline"
                  >
                    مشاهده همه قطعات این گروه
                    <ChevronLeft className="h-3 w-3" aria-hidden="true" />
                  </Link>
                </div>

                {/* لیست زیردسته‌ها */}
                {activeCategory.parts && activeCategory.parts.length > 0 ? (
                  <div className="grid grid-cols-2 gap-x-6 gap-y-2.5">
                    {activeCategory.parts.map((part) => {
                      const partHref = getSubCategoryUrl(activeCategory, part);

                      return (
                        <Link
                          key={part.id}
                          href={partHref}
                          className="py-1 text-sm font-medium text-muted-foreground transition-all duration-150 hover:text-primary hover:-translate-x-1 truncate block"
                        >
                          <span className="truncate">{part.name}</span>
                        </Link>
                      );
                    })}
                  </div>
                ) : (
                  <Typography variant="small" color="muted" className="py-4">
                    هیچ قطعه‌ای برای این دسته‌بندی یافت نشد.
                  </Typography>
                )}
              </div>
            ) : (
              <div className="flex h-full items-center justify-center">
                <Typography variant="small" color="muted">
                  یک دسته‌بندی را از سمت راست انتخاب کنید.
                </Typography>
              </div>
            )}
          </div>
        </>
      )}
    </div>
  );
}

// src/components/sections/Header/components/NavItem/NavItem.tsx

'use client';

import Link from 'next/link';
import { ChevronDown } from 'lucide-react';
import { cn } from '@/design-system/utils/cn';

interface NavItemProps {
  label: string;
  icon?: React.ElementType;
  href: string;
  hasDropdown?: boolean;
  isOpen?: boolean;
  className?: string;
  onMouseEnter?: () => void;
  onMouseLeave?: () => void;
}

export function NavItem({ 
  label, 
  icon: Icon, 
  href, 
  hasDropdown = false, 
  isOpen = false,
  className,
  onMouseEnter,
  onMouseLeave,
}: NavItemProps) {
  return (
    <div 
      className="relative group"
      onMouseEnter={onMouseEnter}
      onMouseLeave={onMouseLeave}
    >
      <Link
        href={href}
        className={cn(
          "flex items-center gap-1 font-medium hover:text-primary transition-colors py-2",
          "text-xs xl:text-sm",
          className
        )}
      >
        {Icon && <Icon className="h-3 w-3 xl:h-4 xl:w-4" />}
        <span className="whitespace-nowrap">{label}</span>
        {hasDropdown && (
          <ChevronDown className={cn('h-3 w-3 transition-transform', isOpen && 'rotate-180')} />
        )}
      </Link>
    </div>
  );
}
