function LottiePlayer({ animationPath }: { animationPath: string }) {
  const containerRef = useRef<HTMLDivElement>(null);
  const [isReady, setIsReady] = useState(false);

  useEffect(() => {
    let animInstance: any = null;
    let isMounted = true;

    const startAnimation = () => {
      const lottie = (window as any).lottie;
      if (lottie && containerRef.current && isMounted) {
        animInstance = lottie.loadAnimation({
          container: containerRef.current,
          renderer: 'svg',
          loop: true,
          autoplay: true,
          path: animationPath,
        });
        setIsReady(true);
      }
    };

    if ((window as any).lottie) {
      startAnimation();
    } else {
      const existingScript = document.getElementById('lottie-cdn-player');
      if (existingScript) {
        existingScript.addEventListener('load', startAnimation);
      } else {
        const script = document.createElement('script');
        script.id = 'lottie-cdn-player';
        script.src = 'https://cdnjs.cloudflare.com/ajax/libs/lottie-web/5.12.2/lottie.min.js';
        script.async = true;
        script.onload = () => {
          if (isMounted) startAnimation();
        };
        document.body.appendChild(script);
      }
    }

    return () => {
      isMounted = false;
      animInstance?.destroy?.();
    };
  }, [animationPath]);

  return (
    <div className="relative w-full h-full flex items-center justify-center">
      <div ref={containerRef} className="w-full h-full max-w-[21.25rem] max-h-[21.25rem]" />
      {!isReady && (
        <span className="sr-only">404 Not Found</span>
      )}
    </div>
  );
}

import type { Metadata } from 'next';

import { getStaticService } from '@/domains/front/static/services/static.service';
import { StaticPageContent } from '@/components/features/Static/StaticPageContent';

interface PageProps {
  params: Promise<{ slug: string }>;
}

export async function generateMetadata({ params }: PageProps): Promise<Metadata> {
  const { slug } = await params;
  const page = await getStaticService().getStaticPage(slug);

  if (!page) {
    return { title: 'صفحه یافت نشد' };
  }

  const meta: Metadata = {
    title: page.seo?.metaTitle || page.title,
    description: page.seo?.metaDescription || undefined,
    keywords: page.seo?.metaKeywords || undefined,
    alternates: page.seo?.canonicalUrl
      ? { canonical: page.seo.canonicalUrl }
      : undefined,
  };

  if (page.noIndex) {
    meta.robots = { index: false, follow: false };
  }

  return meta;
}

export default async function Page({ params }: PageProps) {
  const { slug } = await params;
  const page = await getStaticService().getStaticPage(slug);

  if (!page) {
    notFound();
  }

  return <StaticPageContent slug={slug} initialPage={page} />;
}

const getCachedProductData = cache(async (productCode: number) => {

// استخراج ایمن کد عددی کالا از اسلاگ فارسی/انگلیسی
async function extractProductCode(paramsPromise: ProductPageProps['params']): Promise<number> {
  const resolvedParams = await Promise.resolve(paramsPromise);
  const slugSegment = resolvedParams?.slug?.[0] || '';
  const cleanCode = slugSegment.replace('ykp-', '').replace(/\D/g, '');
  return parseInt(cleanCode, 10);
}

// انکود امن داده‌های اسکیما برای جلوگیری از باگ‌های XSS
function serializeJsonLd(data: unknown): string {
  return JSON.stringify(data).replace(/</g, '\\u003c');
}

/* -------------------------------------------------------------------------- */
/*                              Metadata Generator                            */

export async function generateMetadata({ params }: ProductPageProps): Promise<Metadata> {
  const productCode = await extractProductCode(params);
  if (!productCode || isNaN(productCode)) return { title: 'محصول | یدکچی' };

  try {
    const pageData = await getCachedProductData(productCode);
    if (!pageData?.product) return { title: 'محصول یافت نشد | یدکچی' };

    const product = pageData.product;
    const cleanDesc = (product.seo?.description || product.description || '')
      .replace(/<[^>]*>/g, '')
      .replace(/\s+/g, ' ')
      .trim();

    const resolvedParams = await Promise.resolve(params);
    const safeSlugPath = (resolvedParams.slug || []).map((s) => encodeURIComponent(s)).join('/');
    
    // یکپارچه‌سازی دامنه بدون www با بقیه بخش‌های سایت
    const baseUrl = process.env.NEXT_PUBLIC_SITE_URL || 'https://yadakchi.com';
    const canonicalUrl = `${baseUrl}/product/${safeSlugPath}`;
    const ogImage = product.image ? getFullUrl(product.image) : `${baseUrl}/Logo.svg`;

    const title = `${product.title} | یدکچی`;
    const description = cleanDesc.slice(0, 160) || `خرید اینترنتی ${product.title} با تضمین اصالت و بهترین قیمت در یدکچی`;

    return {
      title,
      description,
      alternates: {
        canonical: canonicalUrl,
      },
      robots: {
        index: true,
        follow: true,
        'max-image-preview': 'large',
        'max-snippet': -1,
      },
      openGraph: {
        title,
        description,
        url: canonicalUrl,
        siteName: 'یدکچی',
        locale: 'fa_IR',
        type: 'website',
        images: [
          {
            url: ogImage,
            width: 800,
            height: 800,
            alt: product.title,
          },
        ],
      },
      twitter: {
        card: 'summary_large_image',
        title,
        description,
        images: [ogImage],
      },
    };
  } catch {
    return { title: 'یدکچی | بازارگاه تخصصی قطعات خودرو' };
  }
}

export default async function ProductPage({ params }: ProductPageProps) {
  const productCode = await extractProductCode(params);

  if (!productCode || isNaN(productCode)) {
    notFound();
  }

  const pageData = await getCachedProductData(productCode);

  if (!pageData || !pageData.product) {
    notFound();
  }

  const queryClient = new QueryClient({
    defaultOptions: {
      queries: {
        staleTime: 5 * 60 * 1000,
      },
    },
  });

  queryClient.setQueryData(['front', 'products', 'page-data', productCode], pageData);

  const product = pageData.product;
  const baseUrl = process.env.NEXT_PUBLIC_SITE_URL || 'https://yadakchi.com';
  const defaultImage = `${baseUrl}/Logo.svg`;

  // گردآوری تمام فروشندگان و استوک‌ها
  const sellers = [
    ...(pageData.shopProducts?.newOnline || []),
    ...(pageData.shopProducts?.newLocal || []),
    ...(pageData.shopProducts?.stockOnline || []),
    ...(pageData.shopProducts?.stockLocal || []),
    ...(pageData.shopProducts?.takeOffOnline || []),
    ...(pageData.shopProducts?.takeOffLocal || []),
  ];

  // محاسبه کمترین و بیشترین قیمت معتبر
  const validPrices = sellers
    .map((s: any) => s.finalPriceRaw || s.price || 0)
    .filter((p: number) => p > 0);

  const lowPrice = validPrices.length > 0 ? Math.min(...validPrices) : 0;
  const highPrice = validPrices.length > 0 ? Math.max(...validPrices) : lowPrice;

  // ساخت لیست تصاویر معتبر با فال‌بک مطمئن (الزام گوگل)
  const productImages =
    product.gallery?.length > 0
      ? product.gallery.map((img: string) => getFullUrl(img))
      : product.image
      ? [getFullUrl(product.image)]
      : [defaultImage];

  // ساخت آفرها با تضمین عدم بروز خطا در گوگل حتی در کالاهای ناموجود
  const offersData =
    sellers.length > 0
      ? {
          '@type': 'AggregateOffer',
          priceCurrency: 'IRR',
          lowPrice: lowPrice,
          highPrice: highPrice,

// تبدیل تاریخ میلادی ISO به تاریخ شمسی با حروف فارسی
function formatToJalali(dateInput: string | Date | null | undefined): string {
  if (!dateInput) return 'ثبت نشده';
  try {
    const d = new Date(dateInput);
    if (isNaN(d.getTime())) return 'ثبت نشده';
    return new Intl.DateTimeFormat('fa-IR', {
      year: 'numeric',
      month: 'long',
      day: 'numeric',
    }).format(d);
  } catch {
    return 'ثبت نشده';
  }
}

// تبدیل ارقام فارسی به انگلیسی جهت فرم
function normalizeNumbers(str: string): string {
  const persianDigits = [/۰/g, /۱/g, /۲/g, /۳/g, /۴/g, /۵/g, /۶/g, /۷/g, /۸/g, /۹/g];
  const arabicDigits = [/٠/g, /١/g, /٢/g, /٣/g, /٤/g, /٥/g, /٦/g, /٧/g, /٨/g, /٩/g];
  
  let result = str;
  for (let i = 0; i < 10; i++) {
    result = result.replace(persianDigits[i], String(i)).replace(arabicDigits[i], String(i));
  }
  return result;
}

function EmptyGarageLottie() {
  const containerRef = useRef<HTMLDivElement>(null);
  const [hasError, setHasError] = useState(false);

  useEffect(() => {
    let anim: any = null;
    let isMounted = true;

    const renderLottie = async () => {
      try {
        let animationData: any = null;
        const paths = ['/anumations/garage-empty.json', '/animations/garage-empty.json'];
        for (const p of paths) {
          try {
            const res = await fetch(p);
            if (res.ok) {
              animationData = await res.json();
              break;
            }
          } catch {}
        }

        if (!animationData || !isMounted || !containerRef.current) {
          if (!animationData) setHasError(true);
          return;
        }

        const lottieModule = await import('lottie-web');
        const lottie = lottieModule.default || lottieModule;

        if (containerRef.current && isMounted) {
          anim = lottie.loadAnimation({
            container: containerRef.current,
            renderer: 'svg',
            loop: true,
            autoplay: true,
            animationData,
          });
        }
      } catch (err) {
        if (isMounted) setHasError(true);
      }
    };

    renderLottie();

    return () => {
      isMounted = false;
      anim?.destroy();
    };
  }, []);

  if (hasError) {
    return (
      <div className="w-32 h-32 flex items-center justify-center rounded-full bg-muted text-muted-foreground">
        <Car className="w-16 h-16" />
      </div>
    );
  }

  return <div ref={containerRef} className="w-64 h-64 max-w-full mx-auto flex items-center justify-center" />;
}

  title: string;
  mileage: number;

}

                    placeholder="روز"
                    value={String(updateDay)}
                    onChange={(e) => setUpdateDay(Number(e.target.value))}
                    options={days}
                  />
                  <Select
                    placeholder="ماه"
                    value={String(updateMonth)}
                    onChange={(e) => setUpdateMonth(Number(e.target.value))}
                    options={months}
                  />
                  <Select
                    placeholder="سال"

                    onChange={(e) => setUpdateYear(Number(e.target.value))}

                  />
                </div>

                  عمر مفید روغن مصرفی *
                </label>
                <div className="grid grid-cols-3 gap-2">
                  {[5000, 7000, 10000].map((limit) => (
                    <button
                      key={limit}
                      type="button"
                      onClick={() => setUpdateOilLimit(limit)}
                      className={cn(
                        "py-2 rounded-xl border text-xs font-semibold transition-all outline-none",
                        updateOilLimit === limit
                          ? "border-foreground bg-muted font-bold text-foreground"
                          : "bg-card text-muted-foreground border-border hover:border-foreground/30 hover:text-foreground"
                      )}
                    >
                      {toPersianDigits(limit.toLocaleString('fa-IR'))} km
                    </button>
                  ))}
                </div>
              </div>

              {/* پیش‌نمایش سرویس بعدی */}
              <div className="w-full bg-muted/40 border border-border/80 rounded-xl p-3 flex items-center justify-between">
                <div className="flex items-center gap-2 text-foreground">
                  <Wrench className="h-4 w-4 shrink-0 text-muted-foreground" />
                  <span className="text-xs font-medium">سرویس بعدی شما:</span>
                </div>
                <div className="flex items-center gap-1 font-bold text-foreground text-sm" dir="ltr">
                  <span>
                    {toPersianDigits(((Number(watchMileage) || 0) + updateOilLimit).toLocaleString('fa-IR'))}
                  </span>
                  <span className="text-2xs font-semibold text-muted-foreground">کیلومتر</span>
                </div>
              </div>
            </form>
          )}
        </ModalBody>

        <div className="flex flex-row gap-2.5 p-5 pt-3 border-t border-border w-full shrink-0">
          <Button
            type="button"
            variant="outline"
            onClick={() => setIsUpdateModalOpen(false)}
            className="flex-1 rounded-xl text-xs h-10 font-medium border-border"
          >
            انصراف
          </Button>
          <Button
            type="submit"
            variant="primary"
            disabled={isFormUnchanged || updateVehicle.isPending}
            isLoading={updateVehicle.isPending}
            className="flex-1 rounded-xl text-xs h-10 font-bold disabled:opacity-50"

          >
            ثبت تغییرات
          </Button>
        </div>
      </Modal>

      <Modal isOpen={isDeleteModalOpen} onClose={() => setIsDeleteModalOpen(false)} className="max-w-md w-full">
        <ModalHeader onClose={() => setIsDeleteModalOpen(false)}>

            <AlertCircle className="h-4 w-4" />
            حذف خودرو از گاراژ
          </ModalTitle>
        </ModalHeader>

        <ModalBody className="p-5 flex flex-col gap-2">
          <p className="text-xs font-bold leading-relaxed text-foreground">
            آیا از حذف «{vehicleToDelete?.car?.model || vehicleToDelete?.carModel || vehicleToDelete?.title}» مطمئن هستید؟
          </p>
          <p className="text-xs text-muted-foreground leading-relaxed">
            با حذف این خودرو، یادآورهای سرویس دوره‌ای و لیست قطعات سازگار با آن پاک خواهند شد.
          </p>
        </ModalBody>

        <div className="flex flex-row gap-2.5 p-5 pt-2 border-t border-border w-full shrink-0">
          <Button
            type="button"
            variant="outline"
            onClick={() => setIsDeleteModalOpen(false)}
            className="flex-1 rounded-xl text-xs h-10 font-medium border-border"
          >
            انصراف
          </Button>
          <Button
            type="button"
            variant="destructive"
            isLoading={deleteVehicle.isPending}
            onClick={confirmDeleteVehicle}
            className="flex-1 rounded-xl text-xs h-10 font-bold"
          >
            بله، حذف شود
          </Button>
        </div>
      </Modal>
    </div>
  );
}
